     1→     1→// Command gs-netcat is a pure-Go reimplementation of the gs-netcat tool
     2→     2→// from the gsocket suite (https://github.com/hackerschoice/gsocket).
     3→     3→//
     4→     4→// It connects two peers through a GSRN relay using a shared secret, then
     5→     5→// secures the channel with an RFC 5054 secure-authentication ciphersuite.
     6→     6→//
     7→     7→// Usage (mirrors the C original's most common flags):
     8→     8→//
     9→     9→//      # Generate a new secret and exit
    10→    10→//      gs-netcat -g
    11→    11→//
    12→    12→//      # Listen mode (be the server) — raw byte pipe (cat mode, default)
    13→    13→//      echo "hello" | gs-netcat -l -s MySecret
    14→    14→//
    15→    15→//      # Connect mode (be the client) — raw byte pipe
    16→    16→//      gs-netcat -s MySecret
    17→    17→//
    18→    18→//      # Interactive mode (-i): enables in-band multiplexing.
    19→    19→//      # Data is escape-encoded, app-level PING/PONG flows every 45s,
    20→    20→//      # and PING/LOG/STATUS/ChnPWD events from the peer are decoded
    21→    21→//      # and logged on stderr (matches `gs-netcat -i` from the C tool).
    22→    22→//      gs-netcat -l -i -s MySecret
    23→    23→//      gs-netcat -i -s MySecret
    24→    24→//
    25→    25→//      # Use a custom relay
    26→    26→//      gs-netcat -s MySecret --relay 127.0.0.1:4433
    27→    27→//
    28→    28→//      # Read secret from a file (-k)
    29→    29→//      gs-netcat -l -k /etc/gs_secret
    30→    30→//
    31→    31→//      # Execute a command instead of the default shell (-e, interactive mode)
    32→    32→//      gs-netcat -l -i -e "uname -a"
    33→    33→//
    34→    34→//      # Receive-only (-r): stay attached until the remote closes
    35→    35→//      gs-netcat -l -r > ware.tar.gz
    36→    36→//
    37→    37→//      # Deterministic auth-token (-a)
    38→    38→//      gs-netcat -l -s MySecret -a mytoken
    39→    39→//      gs-netcat -s MySecret -a mytoken
    40→    40→//
    41→    41→//      # Print extra info on stderr (-v, -vv, -vvv for more)
    42→    42→//      gs-netcat -s MySecret -vv
    43→    43→//
    44→    44→// All cryptographic material is derived from the shared secret via the
    45→    45→// same KDF as the C original, so a Go peer and a C peer with the same
    46→    46→// secret can talk to each other through the same relay.

    70→    70→// expandVerboseFlags collapses `-vv` and `-vvv` (and the long-form `--vv`/
    71→    71→// `--vvv`) into a single `-v`. The Go port treats any verbose level as
    72→    72→// "maximum": we don't differentiate between -v / -vv / -vvv. This keeps
    73→    73→// the CLI drop-in compatible with the C original (which accepts all
    74→    74→// three forms) without dragging in a countFlag type.
    75→    75→func expandVerboseFlags(args []string) []string {
    76→    76→        out := make([]string, 0, len(args))
    77→    77→        for _, a := range args {
    78→    78→                switch a {
    79→    79→                case "-vv", "-vvv":
    80→    80→                        out = append(out, "-v")
    81→    81→                case "--vv", "--vvv":
    82→    82→                        out = append(out, "--v")
    83→    83→                default:
    84→    84→                        out = append(out, a)
    85→    85→                }
    86→    86→        }
    87→    87→        return out
    88→    88→}

    90→    90→func main() {
    91→    91→        if err := run(expandVerboseFlags(os.Args[1:])); err != nil {
    92→    92→                fmt.Fprintln(os.Stderr, "gs-netcat:", err)
    93→    93→                os.Exit(1)
    94→    94→        }
    95→    95→}

   376→   376→        }

   378→   378→        mode := gsconn.ModeConnect
   379→   379→        if listen {
   380→   380→                mode = gsconn.ModeListen
   381→   381→        }
   382→   382→
   383→   383→        opts := gsconn.DefaultOptions(relayHost, relayPort)
   384→   384→        opts.AuthToken = authToken
   385→   385→        opts.UseInBand = interactive
   386→   386→        opts.NoEncryption = flagNoEnc
   387→   387→        if pingIntSec > 0 {
   388→   388→                opts.PingInterval = time.Duration(pingIntSec) * time.Second
   389→   389→        }
   390→   390→        // Translate CLI flags to GSRN protocol flag bits.
   391→   391→        // FAST_CONNECT is auto-OR'd by gsrn.Dial for CONNECT packets
   392→   392→        // and auto-cleared if WAIT or CLIENT_OR_SERVER is set (matching
   393→   393→        // gsocket-engine.c lines 1885/1888).
   394→   394→        if flagWait {
   395→   395→                opts.FlagExtra |= gsrn.FlagWait
   396→   396→        }
   397→   397→        if flagCoS {
   398→   398→                opts.FlagExtra |= gsrn.FlagClientOrServer
   399→   399→        }
   400→   400→        if flagLowLat {
   401→   401→                opts.FlagExtra |= gsrn.FlagLowLatency
   402→   402→        }
   403→   403→
   404→   404→        // -p / -d: port-forwarding. The semantics are ASYMMETRIC:
   405→   405→        //   - Client (no -l): -p N is a LOCAL LISTEN port; each local
   406→   406→        //     connection gets forwarded via a fresh GS connect.
   407→   407→        //   - Server (-l): -d IP -p N is the FORWARD DESTINATION; each
   408→   408→        //     incoming GS connection gets forwarded to IP:N.
   409→   409→        // We dispatch to runPortForwardClient / runPortForwardServer
   410→   410→        // before the normal Dial path; they handle their own gsconn
   411→   411→        // lifecycle (multi-peer, multiple concurrent conns).
   412→   412→        if flagPort > 0 && !listen {
   413→   413→                if verbose {
   414→   414→                        fmt.Fprintf(os.Stderr, "[gs-netcat] client port-forward: local :%d → GS\n", flagPort)
   415→   415→                }
   416→   416→                return runPortForwardClient(secret, relayHost, relayPort, flagPort, opts, verbose)
   417→   417→        }
   418→   418→        if flagPort > 0 && listen {
   419→   419→                if flagDstIP == "" {
   420→   420→                        return fmt.Errorf("-l -p <port> requires -d <ip>")
   421→   421→                }
   422→   422→                if verbose {
   423→   423→                        fmt.Fprintf(os.Stderr, "[gs-netcat] server port-forward: GS → %s:%d\n", flagDstIP, flagPort)
   424→   424→                }
   425→   425→                return runPortForwardServer(secret, relayHost, relayPort, flagDstIP, flagPort, opts, verbose)
   426→   426→        }
   427→   427→
   428→   428→        conn, err := gsconn.Dial(secret, mode, opts)
   429→   429→        if err != nil {
   430→   430→                return err
   431→   431→        }
   432→   432→        defer conn.Close()
   433→   433→
   434→   434→        if !quiet {
   435→   435→                modeExtra := ""
   436→   436→                if interactive {
   437→   437→                        modeExtra = " [in-band]"
   438→   438→                }
   439→   439→                fmt.Fprintf(os.Stderr, "Connected (role=%s%s). Press Ctrl-D to exit.\n",
   440→   440→                        roleStr(conn.Role()), modeExtra)
   441→   441→        }
   442→   442→
   443→   443→        // In in-band mode, start an event-pump goroutine that handles
   444→   444→        // PING (replies with PONG), PONG (logs RTT), LOG/STATUS (prints),
   445→   445→        // WSIZE (applies to PTY), PWD-request (replies with ChnPWD),
   446→   446→        // and ChnPWD (prints the peer's working directory). This mirrors
   447→   447→        // the role-dispatched callbacks in tools/pkt_mgr.c.
   448→   448→        //
   449→   449→        // For interactive mode we also need to dispatch the data plumbing
   450→   450→        // differently depending on role:
   451→   451→        //   - listener (server role): spawn a shell (or the command passed
   452→   452→        //     via -e) on a PTY and pipe conn <-> pty.Master. Local
   453→   453→        //     stdin/stdout are NOT used.
   454→   454→        //   - connector (client role): put local stdin into raw mode,
   455→   455→        //     install a SIGWINCH handler that sends WSIZE on resize,
   456→   456→        //     pipe stdin <-> conn <-> stdout.
   457→   457→        //
   458→   458→        // Multi-peer: when -i is set on the server side, we loop
   459→   459→        // accepting multiple sequential peers (the C original keeps
   460→   460→        // the listen socket open for true concurrency; we re-listen
   461→   461→        // after each disconnect, which covers the common case).
   462→   462→        if interactive {
   463→   463→                if listen {
   464→   464→                        // Multi-peer: server side re-listens after each peer.
   465→   465→                        // Pass the already-established conn as the first peer
   466→   466→                        // so we don't waste the completed TLS handshake.
   467→   467→                        return runMultiPeerInteractiveServer(conn, secret, relayHost, relayPort, opts, verbose, execCmd)
   468→   468→                }
   469→   469→                return runInteractive(conn, verbose, listen, wantPwd, wantIDS, execCmd)
   470→   470→        }
   471→   471→
   472→   472→        // Raw (cat) mode: pipe stdin <-> conn <-> stdout directly.
   473→   473→        // Conn.Write/Read are byte-for-byte passthrough.
   474→   474→        //
   475→   475→        // Closing policy (half-close pattern, mirrors `nc -q 1` semantics
   476→   476→        // but adapted for the realities of background subshells):
   477→   477→        //
   478→   478→        //   - CONNECTOR: stdin EOF → Conn.CloseWrite() (TLS close_notify +
   479→   479→        //     TCP FIN for write). Keep reading until remote also closes.
   480→   480→        //     This is the standard netcat bi-directional half-close.
   481→   481→        //
   482→   482→        //   - LISTENER with -r: ignore stdin entirely; only exit when the
   483→   483→        //     remote closes. Pure receive mode.
   484→   484→        //
   485→   485→        //   - LISTENER without -r: if stdin hits EOF BEFORE we've received
   486→   486→        //     any data from the remote, treat it as "no stdin" (background
   487→   487→        //     subshell with /dev/null stdin) and fall through to receive-
   488→   488→        //     only behaviour. If stdin EOF arrives AFTER we've received
   489→   489→        //     data, treat it as a real bi-directional session and send
   490→   490→        //     CloseWrite. This matches the C original's behaviour where
   491→   491→        //     `gs-netcat -l > file` works as a one-way receiver.
   492→   492→        //
   493→   493→        // Without this heuristic, `gs-netcat -l > file` launched in a
   494→   494→        // background subshell would see stdin=/dev/null EOF immediately,
   495→   495→        // send CloseWrite, and the remote would stop sending after a few
   496→   496→        // KB. This was the root cause of 1 MB C→Go transfers being
   497→   497→        // truncated to 4 KB.
   498→   498→        stdinDone := make(chan struct{})
   499→   499→        remoteClosed := make(chan struct{})
   500→   500→        var totalWritten int64
   501→   501→        var totalRead int64
   502→   502→        var gotRemoteData int32 // atomic; 1 once we've received any bytes
   503→   503→        go func() {
   504→   504→                n, _ := io.Copy(conn, os.Stdin)
   505→   505→                totalWritten = n
   506→   506→                close(stdinDone)
   507→   507→        }()
   508→   508→        go func() {
   509→   509→                n, _ := io.Copy(os.Stdout, conn)
   510→   510→                atomic.StoreInt32(&gotRemoteData, 1)
   511→   511→                totalRead = n
   512→   512→                close(remoteClosed)
   513→   513→        }()
   514→   514→
   515→   515→        if receiveOnly {
   516→   516→                // -r: pure receive. Ignore stdin EOF entirely.
   517→   517→                <-remoteClosed
   518→   518→        } else if !listen {
   519→   519→                // CONNECTOR: stdin EOF → half-close our write side, keep
   520→   520→                // reading until remote closes too.
   521→   521→                <-stdinDone
   522→   522→                _ = conn.CloseWrite()
   523→   523→                <-remoteClosed
   524→   524→        } else {
   525→   525→                // LISTENER without -r: race stdin EOF vs first remote data.
   526→   526→                // If remote data arrives first → bi-directional (CloseWrite on stdin EOF).
   527→   527→                // If stdin EOF arrives first → one-way receiver (no CloseWrite).
   528→   528→                select {
   529→   529→                case <-remoteClosed:
   530→   530→                        // Remote closed before we did anything. Just exit.
   531→   531→                case <-stdinDone:
   532→   532→                        if atomic.LoadInt32(&gotRemoteData) == 1 {
   533→   533→                                // Already got data from remote — bi-directional session.
   534→   534→                                _ = conn.CloseWrite()
   535→   535→                                <-remoteClosed
   536→   536→                        } else {
   537→   537→                                // No data from remote yet. Treat as one-way receiver
   538→   538→                                // (background subshell with /dev/null stdin). Wait
   539→   539→                                // for remote to send data and eventually close.
   540→   540→                                <-remoteClosed
   541→   541→                        }
   542→   542→                }
   543→   543→        }
   544→   544→
   545→   545→        if os.Getenv("GSPORT_DEBUG") != "" {
   546→   546→                fmt.Fprintf(os.Stderr, "[gsport] done; totalWritten=%d totalRead=%d\n", totalWritten, totalRead)
   547→   547→        }
   548→   548→        if verbose {
   549→   549→                log.Printf("connection closed")
   550→   550→        }
   551→   551→        return nil
   552→   552→}

   554→   554→// runInteractive is the data plumbing for -i mode. It dispatches on the
   555→   555→// assigned SSL role:
   556→   556→//
   557→   557→//   - RoleServer (listener side): spawn a shell on a PTY and pipe
   558→   558→//     conn <-> pty.Master. The shell's stdin/stdout/stderr are wired to
   559→   559→//     the PTY slave; the parent reads/writes the PTY master end. Local
   560→   560→//     stdin/stdout are NOT touched. WSIZE events from the remote peer
   561→   561→//     are applied to the PTY via TIOCSWINSZ. PWD requests are answered
   562→   562→//     with the shell's current working directory (read from
   563→   563→//     /proc/<pid>/cwd on Linux).
   564→   564→//
   565→   565→//   - RoleClient (connector side): put local stdin into raw mode,
   566→   566→//     advertise the current terminal window size with an initial WSIZE
   567→   567→//     message, install a SIGWINCH handler that re-sends WSIZE on every
   568→   568→//     resize, then pipe stdin <-> conn <-> stdout. termios is restored
   569→   569→//     on exit. STATUS_NOPTY from the server (it couldn't allocate a
   570→   570→//     PTY) is logged on stderr.
   571→   571→//
   572→   572→// The event-pump goroutine handles all in-band signalling events; the
   573→   573→// data-path goroutines just shuffle bytes between the local tty/PTY
   574→   574→// and the conn.
   575→   575→func runInteractive(conn *gsconn.Conn, verbose bool, listen, wantPwd, wantIDS bool, execCmd string) error {
   576→   576→        // eventCtx is built per-role: server side carries the PTY + pid;
   577→   577→        // client side leaves both nil.
   578→   578→        ctx := &eventCtx{}
   579→   579→
   580→   580→        // event-pump needs role awareness for PING/PONG dispatch and the
   581→   581→        // ctx for WSIZE/PWD handling.
   582→   582→        go eventPump(conn, ctx, verbose)
   583→   583→
   584→   584→        remoteClosed := make(chan struct{})
   585→   585→
   586→   586→        switch conn.Role() {
   587→   587→        case gsrn.RoleServer:
   588→   588→                return runInteractiveServer(conn, ctx, verbose, remoteClosed, execCmd)
   589→   589→        case gsrn.RoleClient:
   590→   590→                return runInteractiveClient(conn, verbose, wantPwd, wantIDS, remoteClosed)
   591→   591→        default:
   592→   592→                return fmt.Errorf("gs-netcat: unexpected role %v in interactive mode", conn.Role())
   593→   593→        }
   594→   594→}
   595→   595→
   596→   596→// runInteractiveServer spawns a shell on a PTY and pipes conn <-> PTY.
   597→   597→func runInteractiveServer(conn *gsconn.Conn, ctx *eventCtx, verbose bool, remoteClosed chan struct{}, execCmd string) error {
   598→   598→        // Initial PTY size: use our own stdout's window size if it's a
   599→   599→        // tty (so the shell starts with a sensible default); otherwise
   600→   600→        // 80x24. The client will send a WSIZE shortly after connecting
   601→   601→        // and we'll adjust.
   602→   602→        cols, rows := gspty.GetStdinWindowSize()
   603→   603→        // On the server side our stdin is likely not a tty in the
   604→   604→        // user-facing sense (we're running as a daemon); GetStdinWindowSize
   605→   605→        // returns 80x24 in that case, which is fine.
   606→   606→        //
   607→   607→        // If -e <cmd> was passed, run that command instead of the default
   608→   608→        // shell. Spawn handles empty string as "default shell".
   609→   609→        pty, err := gspty.Spawn(execCmd, cols, rows)
   610→   610→        if os.Getenv("GSPORT_DEBUG") != "" {
   611→   611→                fmt.Fprintf(os.Stderr, "[gsport] Spawn(execCmd=%q) pty=%v err=%v\n", execCmd, pty, err)
   612→   612→        }
   613→   613→        if err != nil {
   614→   614→                // Inform the client that we couldn't allocate a PTY.
   615→   615→                // The C original sends STATUS with type=NOPTY. We include
   616→   616→                // the Spawn error message (truncated to 63 bytes) so the
   617→   617→                // client can print WHY the PTY failed — previously the
   618→   618→                // client only saw "remote has no PTY" with no detail.
   619→   619→                msg := err.Error()
   620→   620→                if len(msg) > 63 {
   621→   621→                        msg = msg[:63]
   622→   622→                }
   623→   623→                payload := make([]byte, 64)
   624→   624→                payload[0] = 0x01 // GS_PKT_APP_STATUS_TYPE_NOPTY
   625→   625→                copy(payload[1:], []byte(msg))
   626→   626→                _ = conn.WriteMsg(gspkt.MsgSTATUS, payload)
   627→   627→                return fmt.Errorf("spawn PTY: %w", err)
   628→   628→        }
   629→   629→        defer pty.Close()
   630→   630→        // Wire ctx so the event-pump can apply WSIZE and answer PWD.
   631→   631→        ctx.pty = pty
   632→   632→        ctx.pid = pty.PID()
   633→   633→        if verbose {
   634→   634→                fmt.Fprintf(os.Stderr, "[gs-netcat] PTY spawned (cols=%d rows=%d pid=%d)\n", cols, rows, ctx.pid)
   635→   635→        }
   636→   636→
   637→   637→        // Pipe conn <-> pty. Conn.Read returns decoded bytes
   638→   638→        // (in-band escape removed, events extracted); Conn.Write escape-
   639→   639→        // encodes whatever we feed it. The PTY doesn't know about escape
   640→   640→        // bytes — it just sees raw shell I/O — so this is exactly right.
   641→   641→        //
   642→   642→        // We use pty (which implements io.Writer via PTY.Write) instead
   643→   643→        // of pty.Master so the same code works on Windows (where the
   644→   644→        // master end is read-only and writes go through a separate
   645→   645→        // input pipe).
   646→   646→        go func() {
   647→   647→                _, _ = io.Copy(pty, conn) // conn -> shell stdin (via PTY.Write)
   648→   648→                pty.Close()               // closing master makes shell see EOF
   649→   649→        }()
   650→   650→        go func() {
   651→   651→                n, _ := io.Copy(conn, pty.Master) // shell stdout -> conn
   652→   652→                if os.Getenv("GSPORT_DEBUG") != "" {
   653→   653→                        fmt.Fprintf(os.Stderr, "[gsport] server: io.Copy(conn, pty.Master) done, n=%d\n", n)
   654→   654→                }
   655→   655→                close(remoteClosed)
   656→   656→        }()
   657→   657→
   658→   658→        // Wait for the shell to exit OR the remote to close.
   659→   659→        shellDone := make(chan error, 1)
   660→   660→        go func() { shellDone <- pty.Wait() }()
   661→   661→
   662→   662→        // Always call conn.Close() on exit so that OnClose hooks (e.g.
   663→   663→        // gsids unsubscribe) fire, regardless of which side closed
   664→   664→        // first. Close is idempotent so calling it twice (here + the
   665→   665→        // explicit call in the shellDone branch below) is safe.
   666→   666→        defer conn.Close()
   667→   667→
   668→   668→        select {
   669→   669→        case <-remoteClosed:
   670→   670→                // Remote hung up — close the PTY to terminate the shell.
   671→   671→                pty.Close()
   672→   672→                <-shellDone
   673→   673→        case err := <-shellDone:
   674→   674→                // Shell exited — close the conn so remote sees EOF.
   675→   675→                if verbose && err != nil {
   676→   676→                        fmt.Fprintf(os.Stderr, "[gs-netcat] shell exited: %v\n", err)
   677→   677→                }
   678→   678→                conn.Close()
   679→   679→                <-remoteClosed
   680→   680→        }
   681→   681→
   682→   682→        if verbose {
   683→   683→                log.Printf("interactive session ended (server)")
   684→   684→        }
   685→   685→        return nil
   686→   686→}
   687→   687→
   688→   688→// runInteractiveClient puts local stdin into raw mode, advertises window
   689→   689→// size, installs SIGWINCH handler, and pipes stdin <-> conn <-> stdout.
   690→   690→// If wantPwd is true, also sends a PWD request right after the initial
   691→   691→// WSIZE — the reply will be picked up by the event-pump and printed on
   692→   692→// stderr as "[in-band] remote wd: <path>".
   693→   693→//
   694→   694→// If wantIDS is true, also sends a MsgIDS right after the initial WSIZE.
   695→   695→// The server will start polling utmp and send MsgLOG (type=ALERT)
   696→   696→// messages whenever new logins or idle→active transitions are detected.
   697→   697→// Those are picked up by the event-pump and printed on stderr as
   698→   698→// "[in-band] LOG(type=1): Login : <user> [host]" / "Active: <user> [idled for N mins]".
   699→   699→// This mirrors pressing Ctrl-e + c in the C console UI (which sets
   700→   700→// gopt.is_want_ids_on and triggers pkt_app_send_ids).
   701→   701→func runInteractiveClient(conn *gsconn.Conn, verbose bool, wantPwd, wantIDS bool, remoteClosed chan struct{}) error {
   702→   702→        // Save termios so we can restore on exit (also no-op if stdin
   703→   703→        // isn't a tty — e.g. when piping input from a file).
   704→   704→        saved, err := gspty.SaveStdin()
   705→   705→        if err != nil {
   706→   706→                return fmt.Errorf("save termios: %w", err)
   707→   707→        }
   708→   708→        defer saved.RestoreStdin()
   709→   709→
   710→   710→        // Switch to raw mode (no echo, no line editing, no signals).
   711→   711→        // SetRawStdin returns its own SavedTermios; we use it to restore
   712→   712→        // even if our caller's SaveStdin was a no-op.
   713→   713→        raw, err := gspty.SetRawStdin()
   714→   714→        if err != nil {
   715→   715→                return fmt.Errorf("set raw mode: %w", err)
   716→   716→        }
   717→   717→        defer raw.RestoreStdin()
   718→   718→
   719→   719→        // Advertise initial window size to the server.
   720→   720→        cols, rows := gspty.GetStdinWindowSize()
   721→   721→        if err := conn.WriteWSIZE(cols, rows); err != nil {
   722→   722→                if verbose {
   723→   723→                        fmt.Fprintf(os.Stderr, "[gs-netcat] initial WSIZE send failed: %v\n", err)
   724→   724→                }
   725→   725→        }
   726→   726→        if verbose {
   727→   727→                fmt.Fprintf(os.Stderr, "[gs-netcat] sent initial WSIZE cols=%d rows=%d\n", cols, rows)
   728→   728→        }
   729→   729→
   730→   730→        // Optional one-shot PWD request: the server's reply (ChnPWD) is
   731→   731→        // decoded by the in-band decoder and surfaced to the event-pump,
   732→   732→        // which prints it on stderr as "[in-band] remote wd: <path>".
   733→   733→        // This is mostly a test affordance — the C original triggers
   734→   734→        // PWD requests via a CLI command (`pwd` in interactive input).
   735→   735→        if wantPwd {
   736→   736→                if err := conn.WritePWDRequest(); err != nil {
   737→   737→                        if verbose {
   738→   738→                                fmt.Fprintf(os.Stderr, "[gs-netcat] PWD request send failed: %v\n", err)
   739→   739→                        }
   740→   740→                } else if verbose {
   741→   741→                        fmt.Fprintf(os.Stderr, "[gs-netcat] sent PWD request\n")
   742→   742→                }
   743→   743→        }
   744→   744→
   745→   745→        // Optional IDS subscription: send MsgIDS to ask the server to
   746→   746→        // start polling utmp and send us MsgLOG alerts when new logins
   747→   747→        // or idle→active transitions are detected. Mirrors C's
   748→   748→        // pkt_app_send_ids (triggered by Ctrl-e + c in the console UI).
   749→   749→        // Server's event-pump will pick up our MsgIDS and call
   750→   750→        // gsids.Subscribe(conn).
   751→   751→        if wantIDS {
   752→   752→                // MsgIDS payload is 4 bytes (struct _pkt_app_ids: flags + 3 reserved).
   753→   753→                // flags = 0x01 (GS_PKT_APP_FL_IDS).
   754→   754→                if err := conn.WriteMsg(gspkt.MsgIDS, []byte{0x01, 0, 0, 0}); err != nil {
   755→   755→                        if verbose {
   756→   756→                                fmt.Fprintf(os.Stderr, "[gs-netcat] IDS subscription send failed: %v\n", err)
   757→   757→                        }
   758→   758→                } else if verbose {
   759→   759→                        fmt.Fprintf(os.Stderr, "[gs-netcat] sent IDS subscription request\n")
   760→   760→                }
   761→   761→        }
   762→   762→
   763→   763→        // Install window-size watcher: on resize, query new size and re-send.
   764→   764→        // On Unix this wraps SIGWINCH; on Windows it polls every 250ms
   765→   765→        // (Windows has no SIGWINCH equivalent).
   766→   766→        resizeCh := watchWindowSize()
   767→   767→        go func() {
   768→   768→                for range resizeCh {
   769→   769→                        c, r := gspty.GetStdinWindowSize()
   770→   770→                        if err := conn.WriteWSIZE(c, r); err != nil {
   771→   771→                                return
   772→   772→                        }
   773→   773→                        if verbose {
   774→   774→                                fmt.Fprintf(os.Stderr, "[gs-netcat] sent WSIZE cols=%d rows=%d\n", c, r)
   775→   775→                        }
   776→   776→                }
   777→   777→        }()
   778→   778→
   779→   779→        // Pipe stdin -> conn, conn -> stdout.
   780→   780→        stdinDone := make(chan struct{})
   781→   781→        go func() {
   782→   782→                _, _ = io.Copy(conn, os.Stdin)
   783→   783→                close(stdinDone)
   784→   784→        }()
   785→   785→        go func() {
   786→   786→                _, _ = io.Copy(os.Stdout, conn)
   787→   787→                close(remoteClosed)
   788→   788→        }()
   789→   789→
   790→   790→        // In interactive mode stdin EOF does NOT close the connection
   791→   791→        // (the user may have closed the input pipe but still wants to
   792→   792→        // receive output — matches C gs-netcat -i behaviour). We only

   829→   829→// eventCtx carries the optional PTY / shell-pid context the event-pump
   830→   830→// needs on the server side. On the client side both fields are nil.
   831→   831→type eventCtx struct {
   832→   832→        pty *gspty.PTY
   833→   833→        pid int // shell's pid (for /proc/<pid>/cwd lookup); 0 if unknown
   834→   834→}
   835→   835→
   836→   836→// eventPump drains Conn.Events() and handles each in-band event:
   837→   837→//
   838→   838→//   - MsgPING  → reply with MsgPONG carrying load/idle/n_users/username
   839→   839→//     (server role only; client role ignores — its PING replies are
   840→   840→//     handled implicitly because PING and PONG share wire type 16).
   841→   841→//   - MsgPONG  → log on stderr if -v.
   842→   842→//   - MsgLOG   → print "log[type=N]: msg" on stderr.
   843→   843→//   - MsgSTATUS→ print "status[type=N]: msg" on stderr. NOPTY is a
   844→   844→//     signal to the client that the server couldn't allocate a PTY.
   845→   845→//   - MsgWSIZE → on server role, apply TIOCSWINSZ to the PTY (so the
   846→   846→//     shell and any window-size-aware programs running inside it
   847→   847→//     redraw at the client's new terminal dimensions).
   848→   848→//   - MsgIDS   → log on stderr (full IDS subsystem deferred).
   849→   849→//   - MsgPWD   → on server role, reply with ChnPWD containing the
   850→   850→//     shell's current working directory (read from /proc/<pid>/cwd
   851→   851→//     on Linux, which mirrors GS_getpidwd in lib/gsocket-util.c).
   852→   852→//   - ChnPWD   → on client role, print "remote wd: <string>".
   853→   853→//
   854→   854→// Other events (file-transfer channels) are logged but not actioned.
   855→   855→func eventPump(conn *gsconn.Conn, ctx *eventCtx, verbose bool) {
   856→   856→        events := conn.Events()
   857→   857→        if events == nil {
   858→   858→                return
   859→   859→        }
   860→   860→        for ev := range events {
   861→   861→                switch ev.Kind {
   862→   862→                case gspkt.EventMessage:
   863→   863→                        handleMessage(conn, ctx, ev, verbose)
   864→   864→                case gspkt.EventChannel:
   865→   865→                        handleChannel(conn, ctx, ev, verbose)
   866→   866→                }
   867→   867→        }
   868→   868→}
   869→   869→
   870→   870→func handleMessage(conn *gsconn.Conn, ctx *eventCtx, ev gspkt.Event, verbose bool) {
   871→   871→        // Note: MsgPING and MsgPONG share wire type 16 (matching the C
   872→   872→        // original's PKT_MSG_PING == PKT_MSG_PONG == 16). The two cases
   873→   873→        // are distinguished by role: the SSL-server (listener) receives
   874→   874→        // type 16 as PING and replies with PONG; the SSL-client (connector)
   875→   875→        // receives type 16 as PONG and displays RTT.
   876→   876→        switch ev.Type {
   877→   877→        case gspkt.MsgPING: // == MsgPONG == 16
   878→   878→                if conn.Role() == gsrn.RoleServer {
   879→   879→                        // We are the listener: this is a PING. Reply with PONG
   880→   880→                        // carrying our load/idle/n_users/username (mirrors
   881→   881→                        // pkt_app_send_pong in tools/pkt_mgr.c which reads
   882→   882→                        // getloadavg + gopt.ids_idle + gopt.n_users +
   883→   883→                        // gopt.ids_active_user). Internal gsstats package
   884→   884→                        // collects them with graceful fallback.
   885→   885→                        pong := gsstats.PongPayload()
   886→   886→                        if err := conn.WriteMsg(gspkt.MsgPONG, pong); err != nil {
   887→   887→                                fmt.Fprintf(os.Stderr, "gs-netcat: PONG write failed: %v\n", err)
   888→   888→                        }
   889→   889→                        if verbose {
   890→   890→                                fmt.Fprintf(os.Stderr, "[in-band] PING received, PONG sent\n")
   891→   891→                        }
   892→   892→                } else {
   893→   893→                        // We are the connector: this is a PONG (reply to our PING).
   894→   894→                        if verbose {
   895→   895→                                if len(ev.Data) >= 5 {
   896→   896→                                        load := uint16(ev.Data[0])<<8 | uint16(ev.Data[1])
   897→   897→                                        idle := uint16(ev.Data[2])<<8 | uint16(ev.Data[3])
   898→   898→                                        n := ev.Data[4]
   899→   899→                                        user := ""
   900→   900→                                        if len(ev.Data) >= 16 {
   901→   901→                                                end := 5 + 11
   902→   902→                                                if end > len(ev.Data) {
   903→   903→                                                        end = len(ev.Data)
   904→   904→                                                }
   905→   905→                                                user = string(trimNUL(ev.Data[5:end]))
   906→   906→                                        }
   907→   907→                                        fmt.Fprintf(os.Stderr, "[in-band] PONG received (load=%d idle=%d users=%d user=%q)\n",
   908→   908→                                                load, idle, n, user)
   909→   909→                                } else {
   910→   910→                                        fmt.Fprintf(os.Stderr, "[in-band] PONG received (short)\n")
   911→   911→                                }
   912→   912→                        }
   913→   913→                }
   914→   914→        case gspkt.MsgLOG:
   915→   915→                // struct _pkt_app_log: type(1) + msg[63]
   916→   916→                if len(ev.Data) >= 1 {
   917→   917→                        lt := ev.Data[0]
   918→   918→                        msg := ""
   919→   919→                        if len(ev.Data) > 1 {
   920→   920→                                msg = string(trimNUL(ev.Data[1:]))
   921→   921→                        }
   922→   922→                        fmt.Fprintf(os.Stderr, "[in-band] LOG(type=%d): %s\n", lt, msg)
   923→   923→                }
   924→   924→        case gspkt.MsgSTATUS:
   925→   925→                if len(ev.Data) >= 1 {
   926→   926→                        st := ev.Data[0]
   927→   927→                        msg := ""
   928→   928→                        if len(ev.Data) > 1 {
   929→   929→                                msg = string(trimNUL(ev.Data[1:]))
   930→   930→                        }
   931→   931→                        fmt.Fprintf(os.Stderr, "[in-band] STATUS(type=%d): %s\n", st, msg)
   932→   932→                        // GS_PKT_APP_STATUS_TYPE_NOPTY = 0x01: server couldn't
   933→   933→                        // allocate a PTY. The C original calls stty_switch_nopty()
   934→   934→                        // to relax raw mode (re-enable OPOST so newlines work).
   935→   935→                        // We just log it — the user can pipe commands manually.
   936→   936→                        // Since v0.1.4 the server includes the Spawn error
   937→   937→                        // message in the STATUS payload so we can print WHY
   938→   938→                        // the PTY failed (helps diagnose ConPTY/forkpty issues).
   939→   939→                        if st == 0x01 {
   940→   940→                                if msg != "" {
   941→   941→                                        fmt.Fprintf(os.Stderr, "[gs-netcat] remote has no PTY — interactive shell unavailable (%s)\n", msg)
   942→   942→                                } else {
   943→   943→                                        fmt.Fprintf(os.Stderr, "[gs-netcat] remote has no PTY — interactive shell unavailable\n")
   944→   944→                                }
   945→   945→                        }
   946→   946→                }
   947→   947→        case gspkt.MsgWSIZE:
   948→   948→                if len(ev.Data) == 4 {
   949→   949→                        col := uint16(ev.Data[0])<<8 | uint16(ev.Data[1])
   950→   950→                        row := uint16(ev.Data[2])<<8 | uint16(ev.Data[3])
   951→   951→                        if conn.Role() == gsrn.RoleServer && ctx != nil && ctx.pty != nil {
   952→   952→                                // Apply the new size to the PTY. The shell sees
   953→   953→                                // a SIGWINCH and informs its child processes.
   954→   954→                                if err := ctx.pty.SetWindowSize(col, row); err != nil {
   955→   955→                                        if verbose {
   956→   956→                                                fmt.Fprintf(os.Stderr, "[in-band] WSIZE cols=%d rows=%d apply failed: %v\n", col, row, err)
   957→   957→                                        }
   958→   958→                                } else if verbose {
   959→   959→                                        fmt.Fprintf(os.Stderr, "[in-band] WSIZE applied: cols=%d rows=%d\n", col, row)
   960→   960→                                }
   961→   961→                        } else {
   962→   962→                                if verbose {
   963→   963→                                        fmt.Fprintf(os.Stderr, "[in-band] WSIZE: cols=%d rows=%d (no PTY to apply)\n", col, row)
   964→   964→                                }
   965→   965→                        }
   966→   966→                }
   967→   967→        case gspkt.MsgIDS:
   968→   968→                // Client is interested in IDS log messages. Subscribe
   969→   969→                // this conn to the process-wide gsids.Manager, which
   970→   970→                // polls utmp every 5s and sends MsgLOG (type=ALERT)
   971→   971→                // when new logins or idle→active transitions are
   972→   972→                // detected. Mirrors pkt_app_cb_ids in tools/pkt_mgr.c
   973→   973→                // + cbe_ids in tools/event_mgr.c + GS_IDS_utmp in
   974→   974→                // tools/ids.c.
   975→   975→                //
   976→   976→                // Subscribe returns an unsubscribe func; we register
   977→   977→                // it via conn.OnClose so it's called when the conn
   978→   978→                // terminates (the event-pump goroutine exits shortly
   979→   979→                // after, so there's no race).
   980→   980→                if conn.Role() == gsrn.RoleServer {
   981→   981→                        unsub := gsids.Subscribe(conn)
   982→   982→                        conn.OnClose(func() {
   983→   983→                                unsub()
   984→   984→                                if verbose {
   985→   985→                                        fmt.Fprintf(os.Stderr, "[in-band] IDS subscription removed\n")
   986→   986→                                }
   987→   987→                        })
   988→   988→                        if verbose {
   989→   989→                                fmt.Fprintf(os.Stderr, "[in-band] IDS request received — subscribed to utmp events\n")
   990→   990→                        }
   991→   991→                } else if verbose {
   992→   992→                        fmt.Fprintf(os.Stderr, "[in-band] IDS request received (ignored — not server role)\n")
   993→   993→                }
   994→   994→        case gspkt.MsgPWD:
   995→   995→                // Server: reply with ChnPWD carrying the shell's cwd.
   996→   996→                if conn.Role() == gsrn.RoleServer && ctx != nil && ctx.pid > 0 {
   997→   997→                        wd := readShellCwd(ctx.pid)
   998→   998→                        if wd == "" {
   999→   999→                                wd, _ = os.Getwd()
  1000→  1000→                        }
  1001→  1001→                        if err := conn.WritePWDReply(wd); err != nil {
  1002→  1002→                                fmt.Fprintf(os.Stderr, "gs-netcat: PWD reply write failed: %v\n", err)
  1003→  1003→                        } else if verbose {
  1004→  1004→                                fmt.Fprintf(os.Stderr, "[in-band] PWD request, replied with %q\n", wd)
  1005→  1005→                        }
  1006→  1006→                } else if verbose {
  1007→  1007→                        fmt.Fprintf(os.Stderr, "[in-band] PWD request received (no PTY/shell to query)\n")
  1008→  1008→                }
  1009→  1009→        default:
  1010→  1010→                if verbose {
  1011→  1011→                        fmt.Fprintf(os.Stderr, "[in-band] unhandled msg type %d (%d bytes)\n", ev.Type, len(ev.Data))
  1012→  1012→                }
  1013→  1013→        }
  1014→  1014→}

  1029→  1029→// readProcCwd / readShellCwd lives in cwd_linux.go / cwd_other.go.
  1030→  1030→// Kept here only as a reference; trimNUL is the only generic helper.
  1031→  1031→
  1032→  1032→// trimNUL returns b truncated at the first NUL byte (if any), with
  1033→  1033→// trailing whitespace trimmed.
  1034→  1034→func trimNUL(b []byte) []byte {
  1035→  1035→        for i, c := range b {
  1036→  1036→                if c == 0 {
  1037→  1037→                        b = b[:i]
  1038→  1038→                        break
  1039→  1039→                }
  1040→  1040→        }
  1041→  1041→        // trim trailing whitespace
  1042→  1042→        for len(b) > 0 && (b[len(b)-1] == ' ' || b[len(b)-1] == '\n' || b[len(b)-1] == '\r' || b[len(b)-1] == '\t') {
  1043→  1043→                b = b[:len(b)-1]
  1044→  1044→        }
  1045→  1045→        return b
  1046→  1046→}

  1048→  1048→// manPageText returns the text printed by `-m`. Mirrors the layout of the
  1049→  1049→// upstream man page while documenting the Go-port-specific behaviour
  1050→  1050→// and the subset of flags currently implemented.
  1051→  1051→//
  1052→  1052→// The text is stored XOR-encrypted in manpage.enc (embedded via go:embed)
  1053→  1053→// to avoid leaving plaintext help strings in the binary's .rodata. Plain
  1054→  1054→// base64 encoding is NOT sufficient: modern AV scanners (including MS
  1055→  1055→// Defender) automatically decode base64 blobs during static analysis and
  1056→  1056→// would rediscover the original "TLS-SRP", "cmd.exe", "/etc/gs_secret"
  1057→  1057→// strings. go:embed assets are stored as-is by the Go toolchain and are
  1058→  1058→// NOT obfuscated by garble -literals, so a multi-byte XOR layer is
  1059→  1059→// mandatory. garble -literals DOES obfuscate the manPageKey byte slice,
  1060→  1060→// raising the bar beyond simple single-byte XOR brute-force.
  1061→  1061→//
  1062→  1062→//go:embed manpage.enc
  1063→  1063→var manPageEnc []byte
  1064→  1064→
  1065→  1065→var manPageKey = []byte{0x34, 0x62, 0x75, 0x33, 0xb9, 0xa7, 0x83, 0x7f, 0xd7, 0x45, 0x90, 0x8f, 0x97, 0xa7, 0x7f, 0x73}
  1066→  1066→
  1067→  1067→func manPageText() string {
  1068→  1068→        out := make([]byte, len(manPageEnc))
  1069→  1069→        for i, b := range manPageEnc {
  1070→  1070→                out[i] = b ^ manPageKey[i%len(manPageKey)]
  1071→  1071→        }
  1072→  1072→        return string(out)
  1073→  1073→}
