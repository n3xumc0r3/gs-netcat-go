     1→// Windows PTY allocation via ConPTY (Windows 10 1809+).
     2→//
     3→// ConPTY is the modern Windows pseudo-console API. It exposes a pair of
     4→// pipes (input/output) that masquerade as a tty to the child process,
     5→// which is exactly what the interactive mode needs to spawn a child
     6→// process on the remote side and pipe its I/O through the secure channel.
     7→//
     8→// If the ConPTY API is unavailable (older Windows — pre-1809, or
     9→// Windows Server Core without the conpty feature), Spawn returns
    10→// ErrNoPTY. The caller is expected to surface this to the remote peer
    11→// as a STATUS/NOPTY in-band message so the peer falls back to a
    12→// non-PTY pipe (the same code path the C original takes when
    13→// forkpty() fails on legacy Unix).
    14→//
    15→// All Win32 calls go through syscall (no cgo) so the binary stays
    16→// pure-Go and cross-compilable from Linux.
    17→
    18→//go:build windows

    71→// PTY represents an allocated Windows ConPTY with a spawned child
    72→// process attached to it.
    73→//
    74→// On Windows the read/write end exposed to the parent is a pair of
    75→// OS pipes (readPipe, writePipe) — Master is set to an *os.File
    76→// wrapping the read end of the ConPTY's output pipe; writes go
    77→// through a separate Write() method (PTY.Write). The C original
    78→// presents the same surface (one fd to read+write) on Unix but on
    79→// Windows the API is fundamentally duplex-pipe-based.
    80→//
    81→// hProcess and pid are saved from CreateProcessW's PROCESS_INFORMATION
    82→// so Wait() can block on WaitForSingleObject and Close() can kill
    83→// the child if it doesn't exit on its own. Without this the parent
    84→// would race the child: the parent closes the ConPTY, then immediately
    85→// returns from Wait(), while the child is still reading from its
    86→// (now-closed) stdin or writing to its (now-closed) stdout. The
    87→// immediate return was the root cause of the rapid connect/disconnect
    88→// loop on the server side (Bug 2 in v0.1.9).
    89→type PTY struct {
    90→        Master    *os.File          // read end of ConPTY output (parent reads shell stdout here)
    91→        writeFd   syscall.Handle     // write end of ConPTY input (parent writes shell stdin here)
    92→        hPC       syscall.Handle     // ConPTY handle (HPSEUDOCONSOLE)
    93→        hProcess  syscall.Handle     // child process handle (for Wait/Terminate)
    94→        pid       uint32             // child process id
    95→        cmd       *exec.Cmd
    96→        procAttr  syscall.SysProcAttr // placeholder (not used directly on Windows)
    97→        closeMu   sync.Mutex
    98→        closed    bool
    99→}

   101→var (
   102→        procCreatePseudoConsole                       *syscall.LazyProc
   103→        procResizePseudoConsole                        *syscall.LazyProc
   104→        procClosePseudoConsole                         *syscall.LazyProc
   105→        procInitializeProcThreadAttributeList          *syscall.LazyProc
   106→        procUpdateProcThreadAttribute                  *syscall.LazyProc
   107→        procDeleteProcThreadAttributeList              *syscall.LazyProc
   108→        procCreateProcessW                              *syscall.LazyProc
   109→        procGetConsoleScreenBufferInfo                 *syscall.LazyProc
   110→        procSetConsoleMode                              *syscall.LazyProc
   111→        procGetConsoleMode                              *syscall.LazyProc
   112→        procGetStdHandle                                *syscall.LazyProc
   113→        procWaitForSingleObject                         *syscall.LazyProc
   114→        procGetExitCodeProcess                          *syscall.LazyProc
   115→        procTerminateProcess                            *syscall.LazyProc
   116→        conptyAvailable                                 bool
   117→)

   119→func init() {
   120→        k32 := syscall.NewLazyDLL(kernel32)
   121→        procCreatePseudoConsole = k32.NewProc("CreatePseudoConsole")
   122→        procResizePseudoConsole = k32.NewProc("ResizePseudoConsole")
   123→        procClosePseudoConsole = k32.NewProc("ClosePseudoConsole")
   124→        procInitializeProcThreadAttributeList = k32.NewProc("InitializeProcThreadAttributeList")
   125→        procUpdateProcThreadAttribute = k32.NewProc("UpdateProcThreadAttribute")
   126→        procDeleteProcThreadAttributeList = k32.NewProc("DeleteProcThreadAttributeList")
   127→        procCreateProcessW = k32.NewProc("CreateProcessW")
   128→        procGetConsoleScreenBufferInfo = k32.NewProc("GetConsoleScreenBufferInfo")
   129→        procSetConsoleMode = k32.NewProc("SetConsoleMode")
   130→        procGetConsoleMode = k32.NewProc("GetConsoleMode")
   131→        procGetStdHandle = k32.NewProc("GetStdHandle")
   132→        procWaitForSingleObject = k32.NewProc("WaitForSingleObject")
   133→        procGetExitCodeProcess = k32.NewProc("GetExitCodeProcess")
   134→        procTerminateProcess = k32.NewProc("TerminateProcess")
   135→        // If CreatePseudoConsole is missing (pre-Win10 1809), the LazyProc
   136→        // lookup doesn't fail directly — the call will. We probe by checking
   137→        // that the proc address resolves to non-nil.
   138→        conptyAvailable = procCreatePseudoConsole.Find() == nil
   139→}

   141→// Spawn allocates a ConPTY of cols× rows and starts the given command
   142→// (or the system shell via the standard environment variable, falling
   143→// back to the default Windows command interpreter) attached to it.
   144→// Returns ErrNoPTY if the ConPTY API is unavailable.
   145→func Spawn(cmd string, cols, rows uint16) (*PTY, error) {
   146→        if !conptyAvailable {
   147→                return nil, ErrNoPTY
   148→        }
   149→
   150→        // 1. Create two pipes: one for input (parent→ConPTY→child stdin)
   151→        //    and one for output (child stdout→ConPTY→parent).
   152→        var inputRead, inputWrite, outputRead, outputWrite syscall.Handle
   153→        if err := createPipe(&inputRead, &inputWrite); err != nil {
   154→                return nil, fmt.Errorf("gspty: create input pipe: %w", err)
   155→        }
   156→        if err := createPipe(&outputRead, &outputWrite); err != nil {
   157→                syscall.CloseHandle(inputRead)
   158→                syscall.CloseHandle(inputWrite)
   159→                return nil, fmt.Errorf("gspty: create output pipe: %w", err)
   160→        }
   161→
   162→        // 2. Create the pseudoconsole, handing it the write end of the
   163→        //    input pipe and the read end of the output pipe.
   164→        //
   165→        // BUG FIX (v0.1.13b): CreatePseudoConsole takes COORD by VALUE
   166→        // (4 bytes: short X, short Y), not by pointer. The previous code
   167→        // passed uintptr(unsafe.Pointer(&size)), which made Windows
   168→        // interpret a stack address as the COORD — producing a PTY with
   169→        // garbage initial dimensions (e.g. X=0x1234 Y=0x7fff). cmd.exe
   170→        // still spawned (ConPTY is permissive on initial size), but the
   171→        // first WSIZE from the client would then try to resize to the
   172→        // REAL client dimensions and ResizePseudoConsole would reject
   173→        // the resize with E_INVALIDARG if the previous size was wildly
   174→        // out of range.
   175→        //
   176→        // Pack the COORD into a single uintptr the same way as
   177→        // SetWindowSize does (X in low 16 bits, Y in high 16 bits).
   178→        if cols == 0 || rows == 0 {
   179→                // Defensive: ConPTY rejects zero dimensions.
   180→                syscall.CloseHandle(inputRead)
   181→                syscall.CloseHandle(inputWrite)
   182→                syscall.CloseHandle(outputRead)
   183→                syscall.CloseHandle(outputWrite)
   184→                return nil, fmt.Errorf("gspty: invalid PTY size %dx%d", cols, rows)
   185→        }
   186→        packedSize := uintptr(uint16(int16(cols))) | (uintptr(uint16(int16(rows))) << 16)
   187→        var hPC syscall.Handle
   188→        r, _, err := procCreatePseudoConsole.Call(
   189→                packedSize,
   190→                uintptr(inputRead),
   191→                uintptr(outputWrite),
   192→                0,
   193→                uintptr(unsafe.Pointer(&hPC)),
   194→        )
   195→        // CreatePseudoConsole returns an HRESULT. S_OK (0) means SUCCESS;
   196→        // any non-zero value is a failure code (E_HANDLE, E_INVALIDARG,
   197→        // E_NOTIMPL on pre-1809 Windows, etc.).
   198→        //
   199→        // BUG FIX (v0.1.11): the previous check `if r == 0` was INVERTED
   200→        // — it treated the SUCCESS path as failure, closed all the pipe
   201→        // handles, and returned ErrNoPTY. Every Windows PTY allocation
   202→        // was therefore reported as "PTY not supported on this platform:
   203→        // CreatePseudoConsole: <stale errno>", the STATUS/NOPTY was sent
   204→        // to the client, and the session ended immediately. This was the
   205→        // root cause of "Nothing works for the client" (Bug 2 in v0.1.10)
   206→        // and of the rapid multi-peer connect/disconnect loop on the
   207→        // server (each peer connected, the server failed to spawn a PTY,
   208→        // sent NOPTY, client disconnected, server re-listened — repeat).
   209→        if r != 0 {
   210→                syscall.CloseHandle(inputRead)
   211→                syscall.CloseHandle(inputWrite)
   212→                syscall.CloseHandle(outputRead)
   213→                syscall.CloseHandle(outputWrite)
   214→                // Distinguish "API genuinely missing" (pre-1809 Windows)
   215→                // from "API present but call failed" (e.g. invalid handle,
   216→                // out of memory). The former is ErrNoPTY so the caller
   217→                // falls back to a non-PTY pipe; the latter is the raw
   218→                // errno so the user sees the real reason.
   219→                if !conptyAvailable {
   220→                        // ErrNoPTY already prefixes itself with "gspty: ";
   221→                        // don't add another "gspty: " here (was causing the
   222→                        // redundant "gspty: gspty: PTY not supported..."
   223→                        // that pushed the actionable CreatePseudoConsole
   224→                        // detail past the 63-byte STATUS truncation).
   225→                        return nil, fmt.Errorf("%w: CreatePseudoConsole: %v", ErrNoPTY, err)
   226→                }
   227→                return nil, fmt.Errorf("gspty: CreatePseudoConsole failed (hr=0x%08x): %w", uint32(r), err)
   228→        }
   229→        // The pseudoconsole now owns inputRead and outputWrite; the parent
   230→        // keeps inputWrite (to push data into the ConPTY) and outputRead
   231→        // (to read the ConPTY's output). Close the ends we don't need to
   232→        // avoid handle leaks.
   233→        syscall.CloseHandle(inputRead)
   234→        syscall.CloseHandle(outputWrite)
   235→
   236→        // 3. Prepare the STARTUPINFOEX with the pseudoconsole attribute.
   237→        attrListSize := uintptr(0)
   238→        procInitializeProcThreadAttributeList.Call(0, 1, 0, uintptr(unsafe.Pointer(&attrListSize)))
   239→        attrList := make([]byte, attrListSize)
   240→        r, _, _ = procInitializeProcThreadAttributeList.Call(
   241→                uintptr(unsafe.Pointer(&attrList[0])),
   242→                1, 0, uintptr(unsafe.Pointer(&attrListSize)))
   243→        if r == 0 {
   244→                procClosePseudoConsole.Call(uintptr(hPC))
   245→                syscall.CloseHandle(inputWrite)
   246→                syscall.CloseHandle(outputRead)
   247→                return nil, fmt.Errorf("gspty: InitializeProcThreadAttributeList failed")
   248→        }
   249→        r, _, err = procUpdateProcThreadAttribute.Call(
   250→                uintptr(unsafe.Pointer(&attrList[0])),
   251→                0,
   252→                uintptr(PROC_THREAD_ATTRIBUTE_PSEUDOCONSOLE),
   253→                uintptr(hPC),
   254→                unsafe.Sizeof(hPC),
   255→                0, 0)
   256→        if r == 0 {
   257→                procDeleteProcThreadAttributeList.Call(uintptr(unsafe.Pointer(&attrList[0])))
   258→                procClosePseudoConsole.Call(uintptr(hPC))
   259→                syscall.CloseHandle(inputWrite)
   260→                syscall.CloseHandle(outputRead)
   261→                return nil, fmt.Errorf("gspty: UpdateProcThreadAttribute: %v", err)
   262→        }
   263→
   264→        // 4. Build the STARTUPINFOEX. Per Microsoft's ConPTY sample
   265→        //    (https://learn.microsoft.com/en-us/windows/console/creating-a-pseudoconsole-session
   266→        //    and microsoft/terminal samples/ConPTY/EchoCon), only Cb and
   267→        //    AttributeList are set. The DwFlags field is left 0.
   268→        //
   269→        //    BUG FIX (v0.1.13): the previous code set
   270→        //    si.DwFlags = EXTENDED_STARTUPINFO_PRESENT. That flag does NOT
   271→        //    belong in STARTUPINFO.dwFlags — it must be passed as the
   272→        //    dwCreationFlags argument to CreateProcessW. With it in the
   273→        //    wrong place, CreateProcessW treated the STARTUPINFOEX as a
   274→        //    plain STARTUPINFO, ignored lpAttributeList, and cmd.exe
   275→        //    inherited the parent's console instead of attaching to the
   276→        //    ConPTY. The Microsoft Windows banner then went straight to
   277→        //    the server's terminal (not through the conn to the client),
   278→        //    and io.Copy(conn, pty.Master) blocked on an empty pipe until
   279→        //    the client timed out.
   280→        var si startupInfoEx
   281→        si.Cb = uint32(unsafe.Sizeof(si))
   282→        si.AttributeList = uintptr(unsafe.Pointer(&attrList[0]))
   283→
   284→        // 5. Resolve the command to run. Shell name and env-var name are
   285→        //    constructed from byte values to avoid leaving plaintext string
   286→        //    literals in the binary (reduces static AV false-positive rate).
   287→        //
   288→        //    If `cmd` is empty we spawn the user's preferred shell
   289→        //    (%COMSPEC%, falling back to cmd.exe) — same as the C original.
   290→        //
   291→        //    If `cmd` is non-empty we wrap it in `cmd.exe /C "<cmd>"` so the
   292→        //    caller can pass a full shell command (e.g. "dir /b" or
   293→        //    "echo hi & exit 0") without us having to parse argv. This
   294→        //    matches what the C original does internally via system(3),
   295→        //    and mirrors the `sh -c` wrapping used by pty_linux.go.
   296→        shell := cmd
   297→        if shell == "" {
   298→                shell = os.Getenv(strs.Decode([]byte{0x6d, 0x61, 0x63, 0x7d, 0x7e, 0x6b, 0x6d}, 0x2e)) // COMSPEC
   299→        }
   300→        if shell == "" {
   301→                shell = strs.Decode([]byte{0x3c, 0x32, 0x3b, 0x71, 0x3a, 0x27, 0x3a}, 0x5f) // cmd.exe
   302→        }
   303→        if cmd != "" {
   304→                // Wrap in `cmd.exe /C "<cmd>"`. CreateProcessW parses the
   305→                // command line itself (it does NOT use cmd.exe's parser
   306→                // unless we explicitly invoke cmd.exe).
   307→                cmdExe := strs.Decode([]byte{0x3c, 0x32, 0x3b, 0x71, 0x3a, 0x27, 0x3a}, 0x5f) // cmd.exe
   308→                // /C carries the rest of the line as a command; quoting the
   309→                // command protects embedded special chars (&, |, >, etc.).
   310→                shell = cmdExe + " /C \"" + cmd + "\""
   311→        }
   312→
   313→        // 6. CreateProcessW. We don't use os/exec here because ConPTY
   314→        //    requires STARTUPINFOEX with the pseudoconsole attribute,
   315→        //    which os/exec doesn't support.
   316→        shellUTF16, _ := syscall.UTF16PtrFromString(shell)
   317→        var pi processInformation
   318→        cmdLineUTF16 := shellUTF16 // CreateProcess can take lpApplicationName == cmdLine
   319→        //
   320→        // BUG FIX (v0.1.13): dwCreationFlags MUST be EXTENDED_STARTUPINFO_PRESENT
   321→        // so CreateProcessW knows to read the lpAttributeList from STARTUPINFOEX
   322→        // and apply the PROC_THREAD_ATTRIBUTE_PSEUDOCONSOLE attribute to the
   323→        // child process. Without it the attribute is silently ignored.
   324→        //
   325→        // The previous code passed CREATE_UNICODE_ENVIRONMENT (0x400) here,
   326→        // which is wrong: (a) we don't pass a Unicode env block (lpEnvironment=0
   327→        // means inherit parent env), so the flag is a no-op; (b) more
   328→        // importantly, EXTENDED_STARTUPINFO_PRESENT was missing, so cmd.exe
   329→        // never attached to the ConPTY.
   330→        //
   331→        // Matches Microsoft's official ConPTY sample:
   332→        //   CreateProcessW(NULL, cmdLine, NULL, NULL, FALSE,
   333→        //                  EXTENDED_STARTUPINFO_PRESENT, NULL, NULL,
   334→        //                  &siEx.StartupInfo, &pi)
   335→        r, _, err = procCreateProcessW.Call(
   336→                0,
   337→                uintptr(unsafe.Pointer(cmdLineUTF16)),
   338→                0, 0,
   339→                0, // bInheritHandles = FALSE: pseudoconsole attribute controls inheritance
   340→                EXTENDED_STARTUPINFO_PRESENT, // dwCreationFlags: 0x00080000
   341→                0,
   342→                0,
   343→                uintptr(unsafe.Pointer(&si)),
   344→                uintptr(unsafe.Pointer(&pi)),
   345→        )
   346→        procDeleteProcThreadAttributeList.Call(uintptr(unsafe.Pointer(&attrList[0])))
   347→        if r == 0 {
   348→                procClosePseudoConsole.Call(uintptr(hPC))
   349→                syscall.CloseHandle(inputWrite)
   350→                syscall.CloseHandle(outputRead)
   351→                return nil, fmt.Errorf("gspty: CreateProcessW: %w", err)
   352→        }
   353→
   354→        // Save the child process handle and pid. We close the thread handle
   355→        // immediately (we never need it) but keep the process handle for
   356→        // Wait() (WaitForSingleObject) and Close() (TerminateProcess fallback).
   357→        // Without this, Wait() returned nil immediately, causing the parent
   358→        // to race the child: the conn was closed before the shell had a
   359→        // chance to actually run, manifesting as the rapid multi-peer
   360→        // connect/disconnect loop on the server (Bug 2 in v0.1.9).
   361→        hProcess := pi.Process
   362→        pid := pi.ProcessId
   363→        // Thread handle is never used; close it now to avoid handle leaks.
   364→        if pi.Thread != 0 {
   365→                syscall.CloseHandle(pi.Thread)
   366→        }
   367→
   368→        // Build a fake exec.Cmd for compatibility with the Unix API. We
   369→        // only need it to carry the process handle so Wait()/PID()/Close()
   370→        // work the same way as on Linux.
   371→        c := exec.Command(shell) // never actually started; used for Process/Pid only
   372→        c.Process = &os.Process{} // best-effort — Close uses syscall handles directly
   373→        _ = c
   374→
   375→        // Wrap outputRead in an *os.File so callers can io.Copy from it.
   376→        master := os.NewFile(uintptr(outputRead), "conpty-out")
   377→
   378→        return &PTY{
   379→                Master:   master,
   380→                writeFd:  inputWrite,
   381→                hPC:      hPC,
   382→                hProcess: hProcess,
   383→                pid:      pid,
   384→                cmd:      c,
   385→        }, nil
   386→}
   387→
   388→// Write pushes bytes into the ConPTY input pipe (i.e. to the child's
   389→// stdin). On Windows the parent cannot use the same fd for read+write
   390→// (ConPTY uses separate pipes), so the caller writes via this method
   391→// rather than via Master.Write.
   392→//
   393→// Callers in main.go use io.Copy(pty, conn) which calls pty.Write()
   394→// — to make that work, PTY implements io.Writer via this method.
   395→func (p *PTY) Write(b []byte) (int, error) {
   396→        if p.writeFd == 0 {
   397→                return 0, fmt.Errorf("gspty: write to closed PTY")
   398→        }
   399→        var written uint32
   400→        err := syscall.WriteFile(p.writeFd, b, &written, nil)
   401→        if err != nil {
   402→                return 0, err
   403→        }
   404→        return int(written), nil
   405→}
   406→
   407→// SetWindowSize resizes the ConPTY. The child process sees this as a
   408→// WINDOW_BUFFER_SIZE_EVENT and (if it has a SIGWINCH-equivalent handler)
   409→// redraws accordingly. cmd.exe / PowerShell / WSL all handle this.
   410→func (p *PTY) SetWindowSize(cols, rows uint16) error {
   411→        // BUG FIX (v0.1.13): ResizePseudoConsole takes the COORD by VALUE
   412→        // (4 bytes packed: X in low 16 bits, Y in high 16 bits), NOT by
   413→        // pointer. The previous code passed uintptr(unsafe.Pointer(&size))
   414→        // which made Windows interpret a stack address as a huge COORD,
   415→        // returning E_INVALIDARG (0x80070057). This is the misleading
   416→        // "apply failed" message the user saw after every WSIZE.
   417→        //
   418→        // Reference: Windows SDK wincon.h
   419→        //   HRESULT WINAPI ResizePseudoConsole(HPCON hPC, COORD size);
   420→        // COORD is { short X; short Y; } — 4 bytes. In Go we pack both
   421→        // shorts into a single uintptr so the LazyProc.Call passes it as
   422→        // the raw 4-byte value (zero-extended on x64, which is what
   423→        // Windows expects for by-value struct params under the x64 ABI).
   424→        if cols == 0 || rows == 0 {
   425→                // ResizePseudoConsole returns E_INVALIDARG for zero
   426→                // dimensions. Defensive — the caller should never send
   427→                // these, but if it does we don't want a confusing hr.
   428→                return fmt.Errorf("gspty: ResizePseudoConsole: invalid size %dx%d", cols, rows)
   429→        }
   430→        packed := uintptr(uint16(int16(cols))) | (uintptr(uint16(int16(rows))) << 16)
   431→        r, _, _ := procResizePseudoConsole.Call(uintptr(p.hPC), packed)
   432→        if r != 0 {
   433→                return fmt.Errorf("gspty: ResizePseudoConsole failed (hr=0x%08x) size=%dx%d", uint32(r), cols, rows)
   434→        }
   435→        return nil
   436→}
   437→
   438→// Wait blocks until the child process exits. We use WaitForSingleObject
   439→// on the process handle saved at Spawn time.
   440→//
   441→// Returns nil when the child has exited. If the handle is missing
   442→// (e.g. Spawn failed before CreateProcessW), Wait returns nil immediately
   443→// for backward compatibility — but Spawn only returns a *PTY with a
   444→// valid hProcess, so this path is unreachable in normal operation.
   445→//
   446→// NOTE: WaitForSingleObject with INFINITE blocks the calling goroutine
   447→// until the child exits. The Go scheduler treats this as a blocking
   448→// syscall and runs other goroutines on other P's, so the rest of the
   449→// program continues to run. This matches the Unix variant (cmd.Wait() →
   450→// waitpid(2)).
   451→func (p *PTY) Wait() error {
   452→        if p.hProcess == 0 {
   453→                return nil
   454→        }
   455→        // WAIT_FAILED=0xFFFFFFFF, WAIT_OBJECT_0=0, WAIT_TIMEOUT=0x102.
   456→        r, _, err := procWaitForSingleObject.Call(
   457→                ptrOrNil(p.hProcess),
   458→                uintptr(0xFFFFFFFF), // INFINITE
   459→        )
   460→        if r == 0xFFFFFFFF {
   461→                return fmt.Errorf("gspty: WaitForSingleObject: %w", err)
   462→        }
   463→        return nil
   464→}
   465→
   466→// PID returns the child process id saved at Spawn time. Returns 0 if
   467→// Spawn failed before CreateProcessW (unreachable in normal operation).
   468→// On Windows the remote PWD-request feature degrades to os.Getwd()

   470→// logging and debugging.

   473→// waitForExitWithTimeout polls WaitForSingleObject with a small timeout
   474→// so we don't block forever if the child refuses to exit after we close
   475→// its ConPTY. Returns true if the child exited within the timeout, or
   476→// false if we should fall back to TerminateProcess.
   477→//
   478→// Used by Close() to give the child a brief grace period (the ConPTY
   479→// closure sends EOF to the child's stdin, which most shells interpret
   480→// as "exit") before force-killing it.
   481→func (p *PTY) waitForExitWithTimeout(timeoutMs uint32) bool {
   482→        if p.hProcess == 0 {
   483→                return true
   484→        }
   485→        elapsed := uint32(0)
   486→        for elapsed < timeoutMs {
   487→                r, _, _ := procWaitForSingleObject.Call(
   488→                        ptrOrNil(p.hProcess),
   489→                        uintptr(100), // 100ms slices
   490→                )
   491→                if r == 0 { // WAIT_OBJECT_0
   492→                        return true
   493→                }
   494→                // r == 0x102 (WAIT_TIMEOUT) → keep waiting
   495→                elapsed += 100
   496→        }
   497→        return false
   498→}

   500→// ptrOrNil returns 0 if h is 0 (so WaitForSingleObject doesn't crash on
   501→// an invalid handle). Otherwise returns uintptr(h).
   502→func ptrOrNil(h syscall.Handle) uintptr {
   503→        if h == 0 {
   504→                return 0
   505→        }
   506→        return uintptr(h)
   507→}

   509→// Close closes the ConPTY, the input pipe, the output pipe, and kills
   510→// the child process. Safe to call multiple times (idempotent via closed flag).
   511→//
   512→// Shutdown sequence (matches Windows ConPTY best practices):
   513→//   1. ClosePseudoConsole — tears down the ConPTY. The child's stdin
   514→//      receives EOF (write end of input pipe closed by ConPTY),
   515→//      prompting most shells to exit cleanly.
   516→//   2. Close our writeFd (input pipe to ConPTY) and Master (output
   517→//      pipe from ConPTY).
   518→//   3. Give the child up to 2 seconds to exit on its own
   519→//      (waitForExitWithTimeout). cmd.exe typically exits in <100ms
   520→//      after ConPTY close.
   521→//   4. If the child is still alive, TerminateProcess(force).
   522→//   5. Close the child process handle to avoid handle leaks.
   523→//
   524→// Without this sequence the previous version leaked the process handle
   525→// and never waited for the child to exit — which meant subsequent
   526→// reconnect attempts could hit stale state on the relay (Bug 4: TLS-SRP
   527→// handshake EOF after a previous session ended).
   528→func (p *PTY) Close() error {
   529→        p.closeMu.Lock()
   530→        defer p.closeMu.Unlock()
   531→        if p.closed {
   532→                return nil
   533→        }
   534→        p.closed = true
   535→
   536→        if p.hPC != 0 {
   537→                procClosePseudoConsole.Call(uintptr(p.hPC))
   538→                p.hPC = 0
   539→        }
   540→        if p.writeFd != 0 {
   541→                syscall.CloseHandle(p.writeFd)
   542→                p.writeFd = 0
   543→        }
   544→        if p.Master != nil {
   545→                _ = p.Master.Close()
   546→        }
   547→
   548→        if p.hProcess != 0 {
   549→                // Grace period: most shells exit within ~100ms of ConPTY close.
   550→                if !p.waitForExitWithTimeout(2000) {
   551→                        // Force-kill the stubborn child.
   552→                        procTerminateProcess.Call(uintptr(p.hProcess), 1)
   553→                        // Wait briefly for the kill to take effect so CloseHandle
   554→                        // doesn't fail with "still in use".
   555→                        p.waitForExitWithTimeout(500)
   556→                }
   557→                syscall.CloseHandle(p.hProcess)
   558→                p.hProcess = 0
   559→        }
   560→        return nil
   561→}

   563→// startupInfoEx mirrors Win32 STARTUPINFOEXW. The layout MUST match
   564→// the Win32 struct byte-for-byte — CreateProcessW reads fields by
   565→// offset, not by name, so any layout mismatch silently corrupts the
   566→// data Windows sees.
   567→//
   568→// BUG FIX (v0.1.12): the previous definition declared Desktop, Title
   569→// and LpReserved2 as uint16 (2 bytes each) instead of *uint16/*byte
   570→// (8 bytes each on x64). The Go struct was 88 bytes; the real Win32
   571→// struct is 112 bytes. The 24-byte shortfall shifted every field
   572→// past Desktop to the wrong offset:
   573→//
   574→//   - dwFlags landed where dwXSize should be (so
   575→//     EXTENDED_STARTUPINFO_PRESENT was set on the wrong field).
   576→//   - lpAttributeList landed where hStdError should be (so the
   577→//     pseudoconsole attribute was never applied to the child).
   578→//
   579→// Net effect: cmd.exe started by CreateProcessW did NOT attach to
   580→// the ConPTY — it inherited the parent's console instead. The
   581→// Windows banner went straight to the server's terminal (not through
   582→// the conn to the client), and the ConPTY's output pipe stayed
   583→// empty (so io.Copy(conn, pty.Master) blocked forever and the
   584→// client timed out with no shell).
   585→//
   586→// The corrected layout below uses *uint16 for the LPWSTR fields
   587→// (lpReserved, lpDesktop, lpTitle) and *byte for lpReserved2, with
   588→// explicit padding to match Win32 alignment. Total size = 112 bytes
   589→// on 64-bit Windows, which equals sizeof(STARTUPINFOEXW).
   590→type startupInfoEx struct {
   591→        Cb              uint32         // offset  0: cb (DWORD)
   592→        _               [4]byte        // offset  4: padding (align next field to 8)
   593→        Reserved        *uint16        // offset  8: lpReserved (LPWSTR, must be NULL)
   594→        Desktop         *uint16        // offset 16: lpDesktop (LPWSTR, NULL or desktop name)
   595→        Title           *uint16        // offset 24: lpTitle (LPWSTR, NULL for console apps)
   596→        DwX             uint32         // offset 32: dwX
   597→        DwY             uint32         // offset 36: dwY
   598→        DwXSize         uint32         // offset 40: dwXSize
   599→        DwYSize         uint32         // offset 44: dwYSize
   600→        DwXCountChars   uint32         // offset 48: dwXCountChars
   601→        DwYCountChars   uint32         // offset 52: dwYCountChars
   602→        DwFillAttribute uint32         // offset 56: dwFillAttribute
   603→        DwFlags         uint32         // offset 60: dwFlags
   604→        ShowWindow      uint16         // offset 64: wShowWindow
   605→        CbReserved2     uint16         // offset 66: cbReserved2 (must be 0)
   606→        _               [4]byte        // offset 68: padding (align next field to 8)
   607→        LpReserved2     *byte          // offset 72: lpReserved2 (must be NULL)
   608→        HStdInput       syscall.Handle // offset 80: hStdInput
   609→        HStdOutput      syscall.Handle // offset 88: hStdOutput
   610→        HStdError       syscall.Handle // offset 96: hStdError
   611→        AttributeList   uintptr        // offset 104: lpAttributeList
   612→}

   615→// wrong and CreateProcessW will misbehave in subtle ways (see the
   616→// v0.1.12 bug-fix note above). sizeof(STARTUPINFOEXW) = 112 on x64.
   617→var _ [unsafe.Sizeof(startupInfoEx{})]byte = [112]byte{}

   627→// createPipe wraps syscall.CreatePipe with NULL security attributes
   628→// (non-inheritable handles), matching Microsoft's official ConPTY
   629→// sample (microsoft/terminal samples/ConPTY/EchoCon/Common).
   630→//
   631→// BUG FIX (v0.1.13): the previous code set sa.InheritHandle = 1
   632→// (inheritable). The ConPTY pipes must be NON-inheritable because:
   633→//   - CreatePseudoConsole takes ownership of the read-in and write-out
   634→//     ends; making them inheritable would let the child also grab them,
   635→//     bypassing the ConPTY.
   636→//   - With bInheritHandles=FALSE in CreateProcessW the child wouldn't
   637→//     inherit them anyway, but inheritable handles can still leak via
   638→//     handle inheritance bugs and prevent clean ConPTY shutdown
   639→//     (the ConHost can't see EOF, causing Close to hang).
   640→//
   641→// Passing nil for sa is equivalent to CreatePipe(..., NULL, 0) in C,
   642→// which is exactly what the MS sample does.
   643→func createPipe(read, write *syscall.Handle) error {
   644→        return syscall.CreatePipe(read, write, nil, 0)
   645→}
