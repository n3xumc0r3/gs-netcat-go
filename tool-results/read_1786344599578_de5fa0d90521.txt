     1→// Command gs-netcat is a pure-Go reimplementation of the gs-netcat tool
     2→// from the gsocket suite (https://github.com/hackerschoice/gsocket).
     3→//
     4→// It connects two peers through a GSRN relay using a shared secret, then
     5→// secures the channel with an RFC 5054 secure-authentication ciphersuite.
     6→//
     7→// Usage (mirrors the C original's most common flags):
     8→//
     9→//      # Generate a new secret and exit
    10→//      gs-netcat -g
    11→//
    12→//      # Listen mode (be the server) — raw byte pipe (cat mode, default)
    13→//      echo "hello" | gs-netcat -l -s MySecret
    14→//
    15→//      # Connect mode (be the client) — raw byte pipe
    16→//      gs-netcat -s MySecret
    17→//
    18→//      # Interactive mode (-i): enables in-band multiplexing.
    19→//      # Data is escape-encoded, app-level PING/PONG flows every 45s,
    20→//      # and PING/LOG/STATUS/ChnPWD events from the peer are decoded
    21→//      # and logged on stderr (matches `gs-netcat -i` from the C tool).
    22→//      gs-netcat -l -i -s MySecret
    23→//      gs-netcat -i -s MySecret
    24→//
    25→//      # Use a custom relay
    26→//      gs-netcat -s MySecret --relay 127.0.0.1:4433
    27→//
    28→//      # Read secret from a file (-k)
    29→//      gs-netcat -l -k /etc/gs_secret
    30→//
    31→//      # Execute a command instead of the default shell (-e, interactive mode)
    32→//      gs-netcat -l -i -e "uname -a"
    33→//
    34→//      # Receive-only (-r): stay attached until the remote closes
    35→//      gs-netcat -l -r > ware.tar.gz
    36→//
    37→//      # Deterministic auth-token (-a)
    38→//      gs-netcat -l -s MySecret -a mytoken
    39→//      gs-netcat -s MySecret -a mytoken
    40→//
    41→//      # Print extra info on stderr (-v, -vv, -vvv for more)
    42→//      gs-netcat -s MySecret -vv
    43→//
    44→// All cryptographic material is derived from the shared secret via the
    45→// same KDF as the C original, so a Go peer and a C peer with the same
    46→// secret can talk to each other through the same relay.

    72→// expandVerboseFlags collapses `-vv` and `-vvv` (and the long-form `--vv`/
    73→// `--vvv`) into a single `-v`. The Go port treats any verbose level as
    74→// "maximum": we don't differentiate between -v / -vv / -vvv. This keeps
    75→// the CLI drop-in compatible with the C original (which accepts all
    76→// three forms) without dragging in a countFlag type.
    77→func expandVerboseFlags(args []string) []string {
    78→        out := make([]string, 0, len(args))
    79→        for _, a := range args {
    80→                switch a {
    81→                case "-vv", "-vvv":
    82→                        out = append(out, "-v")
    83→                case "--vv", "--vvv":
    84→                        out = append(out, "--v")
    85→                default:
    86→                        out = append(out, a)
    87→                }
    88→        }
    89→        return out
    90→}
    91→
    92→func main() {
    93→        if err := run(expandVerboseFlags(os.Args[1:])); err != nil {
    94→                fmt.Fprintln(os.Stderr, "gs-netcat:", err)
    95→                os.Exit(1)
    96→        }
    97→}

   392→        // Multi-peer token stability fix (v0.1.11):
   393→        //
   394→        // When the server runs in interactive mode (-i -l) the multi-peer
   395→        // loop in multipeer.go re-listens with the relay after each peer
   396→        // disconnects. Each gsconn.Dial call with opts.AuthToken==""
   397→        // generates a NEW random 16-byte wire token (gsconn.go ~line 337).
   398→        // The relay rejects the new listener with BAD_AUTH because the
   399→        // previous listener's slot (keyed by GS-Address + token) is still
   400→        // occupied — manifesting as 2-4 failed reconnect attempts before
   401→        // the relay finally releases the slot.
   402→        //
   403→        // Fix: if the user didn't pass -a (AuthToken is empty) AND we're
   404→        // going to listen interactively, generate ONE random token string
   405→        // per process lifetime and store it in opts.AuthToken. gsconn.Dial
   406→        // will then call gskdf.DeriveAuthToken(opts.AuthToken, address)
   407→        // which is DETERMINISTIC — so every reconnect uses the SAME 16-byte
   408→        // wire token, and the relay sees it as the same listener refreshing
   409→        // its registration rather than a new listener fighting for the slot.
   410→        //
   411→        // This mirrors the C original's behaviour: the listener keeps its
   412→        // SRP identity across reconnects (the C code derives the token from
   413→        // the secret once and never regenerates it for the lifetime of the
   414→        // process).
   415→        if listen && interactive && opts.AuthToken == "" {
   416→                var randBuf [16]byte
   417→                if _, err := rand.Read(randBuf[:]); err == nil {
   418→                        opts.AuthToken = hex.EncodeToString(randBuf[:])
   419→                }
   420→                // If rand.Read fails (extremely unlikely — /dev/urandom
   421→                // unavailable), we leave AuthToken empty and fall back to
   422→                // the per-call random token behaviour. The BAD_AUTH issue
   423→                // will resurface in that case but the tool still works.
   424→        }
   425→        // Translate CLI flags to GSRN protocol flag bits.
   426→        // FAST_CONNECT is auto-OR'd by gsrn.Dial for CONNECT packets
   427→        // and auto-cleared if WAIT or CLIENT_OR_SERVER is set (matching
   428→        // gsocket-engine.c lines 1885/1888).
   429→        if flagWait {
   430→                opts.FlagExtra |= gsrn.FlagWait
   431→        }
   432→        if flagCoS {
   433→                opts.FlagExtra |= gsrn.FlagClientOrServer
   434→        }
   435→        if flagLowLat {
   436→                opts.FlagExtra |= gsrn.FlagLowLatency
   437→        }

   439→        // -p / -d: port-forwarding. The semantics are ASYMMETRIC:
   440→        //   - Client (no -l): -p N is a LOCAL LISTEN port; each local
   441→        //     connection gets forwarded via a fresh GS connect.
   442→        //   - Server (-l): -d IP -p N is the FORWARD DESTINATION; each
   443→        //     incoming GS connection gets forwarded to IP:N.
   444→        // We dispatch to runPortForwardClient / runPortForwardServer
   445→        // before the normal Dial path; they handle their own gsconn
   446→        // lifecycle (multi-peer, multiple concurrent conns).
   447→        if flagPort > 0 && !listen {
   448→                if verbose {
   449→                        fmt.Fprintf(os.Stderr, "[gs-netcat] client port-forward: local :%d → GS\n", flagPort)
   450→                }
   451→                return runPortForwardClient(secret, relayHost, relayPort, flagPort, opts, verbose)
   452→        }
   453→        if flagPort > 0 && listen {
   454→                if flagDstIP == "" {
   455→                        return fmt.Errorf("-l -p <port> requires -d <ip>")
   456→                }
   457→                if verbose {
   458→                        fmt.Fprintf(os.Stderr, "[gs-netcat] server port-forward: GS → %s:%d\n", flagDstIP, flagPort)
   459→                }
   460→                return runPortForwardServer(secret, relayHost, relayPort, flagDstIP, flagPort, opts, verbose)
   461→        }
   462→
   463→        conn, err := gsconn.Dial(secret, mode, opts)
   464→        if err != nil {
   465→                return err
   466→        }
   467→        defer conn.Close()
   468→
   469→        if !quiet {
   470→                modeExtra := ""
   471→                if interactive {
   472→                        modeExtra = " [in-band]"
   473→                }
   474→                fmt.Fprintf(os.Stderr, "Connected (role=%s%s). Press Ctrl-D to exit.\n",
   475→                        roleStr(conn.Role()), modeExtra)
   476→        }
   477→
   478→        // In in-band mode, start an event-pump goroutine that handles
   479→        // PING (replies with PONG), PONG (logs RTT), LOG/STATUS (prints),
   480→        // WSIZE (applies to PTY), PWD-request (replies with ChnPWD),
   481→        // and ChnPWD (prints the peer's working directory). This mirrors
   482→        // the role-dispatched callbacks in tools/pkt_mgr.c.
   483→        //
   484→        // For interactive mode we also need to dispatch the data plumbing
   485→        // differently depending on role:
   486→        //   - listener (server role): spawn a shell (or the command passed
   487→        //     via -e) on a PTY and pipe conn <-> pty.Master. Local
   488→        //     stdin/stdout are NOT used.
   489→        //   - connector (client role): put local stdin into raw mode,
   490→        //     install a SIGWINCH handler that sends WSIZE on resize,
   491→        //     pipe stdin <-> conn <-> stdout.
   492→        //
   493→        // Multi-peer: when -i is set on the server side, we loop
   494→        // accepting multiple sequential peers (the C original keeps
   495→        // the listen socket open for true concurrency; we re-listen
   496→        // after each disconnect, which covers the common case).
   497→        if interactive {
   498→                if listen {
   499→                        // Multi-peer: server side re-listens after each peer.
   500→                        // Pass the already-established conn as the first peer
   501→                        // so we don't waste the completed TLS handshake.
   502→                        return runMultiPeerInteractiveServer(conn, secret, relayHost, relayPort, opts, verbose, execCmd)
   503→                }
   504→                return runInteractive(conn, verbose, listen, wantPwd, wantIDS, execCmd)
   505→        }
   506→
   507→        // Raw (cat) mode: pipe stdin <-> conn <-> stdout directly.
   508→        // Conn.Write/Read are byte-for-byte passthrough.
   509→        //
   510→        // Closing policy (half-close pattern, mirrors `nc -q 1` semantics
   511→        // but adapted for the realities of background subshells):
   512→        //
   513→        //   - CONNECTOR: stdin EOF → Conn.CloseWrite() (TLS close_notify +
   514→        //     TCP FIN for write). Keep reading until remote also closes.
   515→        //     This is the standard netcat bi-directional half-close.
   516→        //
   517→        //   - LISTENER with -r: ignore stdin entirely; only exit when the
   518→        //     remote closes. Pure receive mode.
   519→        //
   520→        //   - LISTENER without -r: if stdin hits EOF BEFORE we've received
   521→        //     any data from the remote, treat it as "no stdin" (background
   522→        //     subshell with /dev/null stdin) and fall through to receive-
   523→        //     only behaviour. If stdin EOF arrives AFTER we've received
   524→        //     data, treat it as a real bi-directional session and send
   525→        //     CloseWrite. This matches the C original's behaviour where
   526→        //     `gs-netcat -l > file` works as a one-way receiver.
   527→        //
   528→        // Without this heuristic, `gs-netcat -l > file` launched in a
   529→        // background subshell would see stdin=/dev/null EOF immediately,
   530→        // send CloseWrite, and the remote would stop sending after a few
   531→        // KB. This was the root cause of 1 MB C→Go transfers being
   532→        // truncated to 4 KB.
   533→        stdinDone := make(chan struct{})
   534→        remoteClosed := make(chan struct{})
   535→        var totalWritten int64
   536→        var totalRead int64
   537→        var gotRemoteData int32 // atomic; 1 once we've received any bytes
   538→        go func() {
   539→                n, _ := io.Copy(conn, os.Stdin)
   540→                totalWritten = n
   541→                close(stdinDone)
   542→        }()
   543→        go func() {
   544→                n, _ := io.Copy(os.Stdout, conn)
   545→                atomic.StoreInt32(&gotRemoteData, 1)
   546→                totalRead = n
   547→                close(remoteClosed)
   548→        }()
   549→
   550→        if receiveOnly {
   551→                // -r: pure receive. Ignore stdin EOF entirely.
   552→                <-remoteClosed
   553→        } else if !listen {
   554→                // CONNECTOR: stdin EOF → half-close our write side, keep
   555→                // reading until remote closes too.
   556→                <-stdinDone
   557→                _ = conn.CloseWrite()
   558→                <-remoteClosed
   559→        } else {
   560→                // LISTENER without -r: race stdin EOF vs first remote data.
   561→                // If remote data arrives first → bi-directional (CloseWrite on stdin EOF).
   562→                // If stdin EOF arrives first → one-way receiver (no CloseWrite).
   563→                select {
   564→                case <-remoteClosed:
   565→                        // Remote closed before we did anything. Just exit.
   566→                case <-stdinDone:
   567→                        if atomic.LoadInt32(&gotRemoteData) == 1 {
   568→                                // Already got data from remote — bi-directional session.
   569→                                _ = conn.CloseWrite()
   570→                                <-remoteClosed
   571→                        } else {
   572→                                // No data from remote yet. Treat as one-way receiver
   573→                                // (background subshell with /dev/null stdin). Wait
   574→                                // for remote to send data and eventually close.
   575→                                <-remoteClosed
   576→                        }
   577→                }
   578→        }
   579→
   580→        if os.Getenv("GSPORT_DEBUG") != "" {
   581→                fmt.Fprintf(os.Stderr, "[gsport] done; totalWritten=%d totalRead=%d\n", totalWritten, totalRead)
   582→        }
   583→        if verbose {
   584→                log.Printf("connection closed")
   585→        }
   586→        return nil
   587→}

   589→// runInteractive is the data plumbing for -i mode. It dispatches on the
   590→// assigned SSL role:
   591→//
   592→//   - RoleServer (listener side): spawn a shell on a PTY and pipe
   593→//     conn <-> pty.Master. The shell's stdin/stdout/stderr are wired to
   594→//     the PTY slave; the parent reads/writes the PTY master end. Local
   595→//     stdin/stdout are NOT touched. WSIZE events from the remote peer
   596→//     are applied to the PTY via TIOCSWINSZ. PWD requests are answered
   597→//     with the shell's current working directory (read from
   598→//     /proc/<pid>/cwd on Linux).
   599→//
   600→//   - RoleClient (connector side): put local stdin into raw mode,
   601→//     advertise the current terminal window size with an initial WSIZE
   602→//     message, install a SIGWINCH handler that re-sends WSIZE on every
   603→//     resize, then pipe stdin <-> conn <-> stdout. termios is restored
   604→//     on exit. STATUS_NOPTY from the server (it couldn't allocate a
   605→//     PTY) is logged on stderr.
   606→//
   607→// The event-pump goroutine handles all in-band signalling events; the
   608→// data-path goroutines just shuffle bytes between the local tty/PTY
   609→// and the conn.
   610→func runInteractive(conn *gsconn.Conn, verbose bool, listen, wantPwd, wantIDS bool, execCmd string) error {
   611→        // eventCtx is built per-role: server side carries the PTY + pid;
   612→        // client side leaves both nil.
   613→        ctx := &eventCtx{}
   614→
   615→        // event-pump needs role awareness for PING/PONG dispatch and the
   616→        // ctx for WSIZE/PWD handling.
   617→        go eventPump(conn, ctx, verbose)
   618→
   619→        remoteClosed := make(chan struct{})
   620→
   621→        switch conn.Role() {
   622→        case gsrn.RoleServer:
   623→                return runInteractiveServer(conn, ctx, verbose, remoteClosed, execCmd)
   624→        case gsrn.RoleClient:
   625→                return runInteractiveClient(conn, verbose, wantPwd, wantIDS, remoteClosed)
   626→        default:
   627→                return fmt.Errorf("gs-netcat: unexpected role %v in interactive mode", conn.Role())
   628→        }
   629→}

   631→// runInteractiveServer spawns a shell on a PTY and pipes conn <-> PTY.
   632→func runInteractiveServer(conn *gsconn.Conn, ctx *eventCtx, verbose bool, remoteClosed chan struct{}, execCmd string) error {
   633→        // Initial PTY size: use our own stdout's window size if it's a
   634→        // tty (so the shell starts with a sensible default); otherwise
   635→        // 80x24. The client will send a WSIZE shortly after connecting
   636→        // and we'll adjust.
   637→        cols, rows := gspty.GetStdinWindowSize()
   638→        // On the server side our stdin is likely not a tty in the
   639→        // user-facing sense (we're running as a daemon); GetStdinWindowSize
   640→        // returns 80x24 in that case, which is fine.
   641→        //
   642→        // If -e <cmd> was passed, run that command instead of the default
   643→        // shell. Spawn handles empty string as "default shell".
   644→        pty, err := gspty.Spawn(execCmd, cols, rows)
   645→        if os.Getenv("GSPORT_DEBUG") != "" {
   646→                fmt.Fprintf(os.Stderr, "[gsport] Spawn(execCmd=%q) pty=%v err=%v\n", execCmd, pty, err)
   647→        }
   648→        if err != nil {
   649→                // Inform the client that we couldn't allocate a PTY.
   650→                // The C original sends STATUS with type=NOPTY. We include
   651→                // the Spawn error message (truncated to 63 bytes — the
   652→                // MsgSTATUS protocol limit) so the client can print WHY
   653→                // the PTY failed — previously the client only saw "remote
   654→                // has no PTY" with no detail.
   655→                //
   656→                // We strip any leading "gspty: " prefix from the message
   657→                // before truncation so the actionable part of the error
   658→                // (e.g. "CreatePseudoConsole failed (hr=0x...): ...")
   659→                // survives the 63-byte cut. Without this, the previous
   660→                // "gspty: gspty: PTY not supported on this platform:
   661→                // CreatePseudoConsole: ..." string was truncated to
   662→                // "gspty: gspty: PTY not supported on this platform:
   663→                // CreatePseudoC" — losing the actual error code.
   664→                msg := err.Error()
   665→                for strings.HasPrefix(msg, "gspty: ") {
   666→                        msg = strings.TrimPrefix(msg, "gspty: ")
   667→                }
   668→                if len(msg) > 63 {
   669→                        msg = msg[:63]
   670→                }
   671→                payload := make([]byte, 64)
   672→                payload[0] = 0x01 // GS_PKT_APP_STATUS_TYPE_NOPTY
   673→                copy(payload[1:], []byte(msg))
   674→                _ = conn.WriteMsg(gspkt.MsgSTATUS, payload)
   675→                return fmt.Errorf("spawn PTY: %w", err)
   676→        }
   677→        defer pty.Close()
   678→        // Wire ctx so the event-pump can apply WSIZE and answer PWD.
   679→        ctx.pty = pty
   680→        ctx.pid = pty.PID()
   681→        if verbose {
   682→                fmt.Fprintf(os.Stderr, "[gs-netcat] PTY spawned (cols=%d rows=%d pid=%d)\n", cols, rows, ctx.pid)
   683→        }
   684→
   685→        // Pipe conn <-> pty. Conn.Read returns decoded bytes
   686→        // (in-band escape removed, events extracted); Conn.Write escape-
   687→        // encodes whatever we feed it. The PTY doesn't know about escape
   688→        // bytes — it just sees raw shell I/O — so this is exactly right.
   689→        //
   690→        // We use pty (which implements io.Writer via PTY.Write) instead
   691→        // of pty.Master so the same code works on Windows (where the
   692→        // master end is read-only and writes go through a separate
   693→        // input pipe).
   694→        go func() {
   695→                _, _ = io.Copy(pty, conn) // conn -> shell stdin (via PTY.Write)
   696→                pty.Close()               // closing master makes shell see EOF
   697→        }()
   698→        go func() {
   699→                n, _ := io.Copy(conn, pty.Master) // shell stdout -> conn
   700→                if os.Getenv("GSPORT_DEBUG") != "" {
   701→                        fmt.Fprintf(os.Stderr, "[gsport] server: io.Copy(conn, pty.Master) done, n=%d\n", n)
   702→                }
   703→                close(remoteClosed)
   704→        }()
   705→
   706→        // Wait for the shell to exit OR the remote to close.
   707→        shellDone := make(chan error, 1)
   708→        go func() { shellDone <- pty.Wait() }()
   709→
   710→        // Always call conn.Close() on exit so that OnClose hooks (e.g.
   711→        // gsids unsubscribe) fire, regardless of which side closed
   712→        // first. Close is idempotent so calling it twice (here + the
   713→        // explicit call in the shellDone branch below) is safe.
   714→        defer conn.Close()
   715→
   716→        select {
   717→        case <-remoteClosed:
   718→                // Remote hung up — close the PTY to terminate the shell.
   719→                pty.Close()
   720→                <-shellDone
   721→        case err := <-shellDone:
   722→                // Shell exited — close the conn so remote sees EOF.
   723→                if verbose && err != nil {
   724→                        fmt.Fprintf(os.Stderr, "[gs-netcat] shell exited: %v\n", err)
   725→                }
   726→                conn.Close()
   727→                <-remoteClosed
   728→        }
   729→
   730→        if verbose {
   731→                log.Printf("interactive session ended (server)")
   732→        }
   733→        return nil
   734→}
   735→
   736→// runInteractiveClient puts local stdin into raw mode, advertises window
   737→// size, installs SIGWINCH handler, and pipes stdin <-> conn <-> stdout.
   738→// If wantPwd is true, also sends a PWD request right after the initial
   739→// WSIZE — the reply will be picked up by the event-pump and printed on
   740→// stderr as "[in-band] remote wd: <path>".
   741→//
   742→// If wantIDS is true, also sends a MsgIDS right after the initial WSIZE.
   743→// The server will start polling utmp and send MsgLOG (type=ALERT)
   744→// messages whenever new logins or idle→active transitions are detected.
   745→// Those are picked up by the event-pump and printed on stderr as
   746→// "[in-band] LOG(type=1): Login : <user> [host]" / "Active: <user> [idled for N mins]".
   747→// This mirrors pressing Ctrl-e + c in the C console UI (which sets
   748→// gopt.is_want_ids_on and triggers pkt_app_send_ids).
   749→func runInteractiveClient(conn *gsconn.Conn, verbose bool, wantPwd, wantIDS bool, remoteClosed chan struct{}) error {
   750→        // Save termios so we can restore on exit (also no-op if stdin
   751→        // isn't a tty — e.g. when piping input from a file).
   752→        saved, err := gspty.SaveStdin()
   753→        if err != nil {
   754→                return fmt.Errorf("save termios: %w", err)
   755→        }
   756→        defer saved.RestoreStdin()
   757→
   758→        // Switch to raw mode (no echo, no line editing, no signals).
   759→        // SetRawStdin returns its own SavedTermios; we use it to restore
   760→        // even if our caller's SaveStdin was a no-op.
   761→        raw, err := gspty.SetRawStdin()
   762→        if err != nil {
   763→                return fmt.Errorf("set raw mode: %w", err)
   764→        }
   765→        defer raw.RestoreStdin()
   766→
   767→        // Advertise initial window size to the server.
   768→        cols, rows := gspty.GetStdinWindowSize()
   769→        if err := conn.WriteWSIZE(cols, rows); err != nil {
   770→                if verbose {
   771→                        fmt.Fprintf(os.Stderr, "[gs-netcat] initial WSIZE send failed: %v\n", err)
   772→                }
   773→        }
   774→        if verbose {
   775→                fmt.Fprintf(os.Stderr, "[gs-netcat] sent initial WSIZE cols=%d rows=%d\n", cols, rows)
   776→        }
   777→
   778→        // Optional one-shot PWD request: the server's reply (ChnPWD) is
   779→        // decoded by the in-band decoder and surfaced to the event-pump,
   780→        // which prints it on stderr as "[in-band] remote wd: <path>".
   781→        // This is mostly a test affordance — the C original triggers
   782→        // PWD requests via a CLI command (`pwd` in interactive input).
   783→        if wantPwd {
   784→                if err := conn.WritePWDRequest(); err != nil {
   785→                        if verbose {
   786→                                fmt.Fprintf(os.Stderr, "[gs-netcat] PWD request send failed: %v\n", err)
   787→                        }
   788→                } else if verbose {
   789→                        fmt.Fprintf(os.Stderr, "[gs-netcat] sent PWD request\n")
   790→                }
   791→        }
   792→
   793→        // Optional IDS subscription: send MsgIDS to ask the server to
   794→        // start polling utmp and send us MsgLOG alerts when new logins
   795→        // or idle→active transitions are detected. Mirrors C's
   796→        // pkt_app_send_ids (triggered by Ctrl-e + c in the console UI).
   797→        // Server's event-pump will pick up our MsgIDS and call
   798→        // gsids.Subscribe(conn).
   799→        if wantIDS {
   800→                // MsgIDS payload is 4 bytes (struct _pkt_app_ids: flags + 3 reserved).
   801→                // flags = 0x01 (GS_PKT_APP_FL_IDS).
   802→                if err := conn.WriteMsg(gspkt.MsgIDS, []byte{0x01, 0, 0, 0}); err != nil {
   803→                        if verbose {
   804→                                fmt.Fprintf(os.Stderr, "[gs-netcat] IDS subscription send failed: %v\n", err)
   805→                        }
   806→                } else if verbose {
   807→                        fmt.Fprintf(os.Stderr, "[gs-netcat] sent IDS subscription request\n")
   808→                }
   809→        }
   810→
   811→        // Install window-size watcher: on resize, query new size and re-send.
   812→        // On Unix this wraps SIGWINCH; on Windows it polls every 250ms
   813→        // (Windows has no SIGWINCH equivalent).
   814→        resizeCh := watchWindowSize()
   815→        go func() {
   816→                for range resizeCh {
   817→                        c, r := gspty.GetStdinWindowSize()
   818→                        if err := conn.WriteWSIZE(c, r); err != nil {
   819→                                return
   820→                        }
   821→                        if verbose {
   822→                                fmt.Fprintf(os.Stderr, "[gs-netcat] sent WSIZE cols=%d rows=%d\n", c, r)
   823→                        }
   824→                }
   825→        }()
   826→
   827→        // Pipe stdin -> conn, conn -> stdout.
   828→        stdinDone := make(chan struct{})
   829→        go func() {
   830→                _, _ = io.Copy(conn, os.Stdin)
   831→                close(stdinDone)
   832→        }()
   833→        go func() {
   834→                _, _ = io.Copy(os.Stdout, conn)
   835→                close(remoteClosed)
   836→        }()
   837→
   838→        // In interactive mode stdin EOF does NOT close the connection
   839→        // (the user may have closed the input pipe but still wants to
   840→        // receive output — matches C gs-netcat -i behaviour). We only
   841→        // exit on remote-closed.
   842→        <-remoteClosed
   843→
   844→        if verbose {
   845→                log.Printf("interactive session ended (client)")
   846→        }
   847→        return nil
   848→}

   877→// eventCtx carries the optional PTY / shell-pid context the event-pump
   878→// needs on the server side. On the client side both fields are nil.
   879→type eventCtx struct {
   880→        pty *gspty.PTY
   881→        pid int // shell's pid (for /proc/<pid>/cwd lookup); 0 if unknown
   882→}
   883→
   884→// eventPump drains Conn.Events() and handles each in-band event:
   885→//
   886→//   - MsgPING  → reply with MsgPONG carrying load/idle/n_users/username
   887→//     (server role only; client role ignores — its PING replies are
   888→//     handled implicitly because PING and PONG share wire type 16).
   889→//   - MsgPONG  → log on stderr if -v.
   890→//   - MsgLOG   → print "log[type=N]: msg" on stderr.
   891→//   - MsgSTATUS→ print "status[type=N]: msg" on stderr. NOPTY is a
   892→//     signal to the client that the server couldn't allocate a PTY.
   893→//   - MsgWSIZE → on server role, apply TIOCSWINSZ to the PTY (so the
   894→//     shell and any window-size-aware programs running inside it
   895→//     redraw at the client's new terminal dimensions).
   896→//   - MsgIDS   → log on stderr (full IDS subsystem deferred).
   897→//   - MsgPWD   → on server role, reply with ChnPWD containing the
   898→//     shell's current working directory (read from /proc/<pid>/cwd
   899→//     on Linux, which mirrors GS_getpidwd in lib/gsocket-util.c).
   900→//   - ChnPWD   → on client role, print "remote wd: <string>".
   901→//
   902→// Other events (file-transfer channels) are logged but not actioned.
   903→func eventPump(conn *gsconn.Conn, ctx *eventCtx, verbose bool) {
   904→        events := conn.Events()
   905→        if events == nil {
   906→                return
   907→        }
   908→        for ev := range events {
   909→                switch ev.Kind {
   910→                case gspkt.EventMessage:
   911→                        handleMessage(conn, ctx, ev, verbose)
   912→                case gspkt.EventChannel:
   913→                        handleChannel(conn, ctx, ev, verbose)
   914→                }
   915→        }
   916→}
   917→
   918→func handleMessage(conn *gsconn.Conn, ctx *eventCtx, ev gspkt.Event, verbose bool) {
   919→        // Note: MsgPING and MsgPONG share wire type 16 (matching the C
   920→        // original's PKT_MSG_PING == PKT_MSG_PONG == 16). The two cases
   921→        // are distinguished by role: the SSL-server (listener) receives
   922→        // type 16 as PING and replies with PONG; the SSL-client (connector)
   923→        // receives type 16 as PONG and displays RTT.
   924→        switch ev.Type {
   925→        case gspkt.MsgPING: // == MsgPONG == 16
   926→                if conn.Role() == gsrn.RoleServer {
   927→                        // We are the listener: this is a PING. Reply with PONG
   928→                        // carrying our load/idle/n_users/username (mirrors
   929→                        // pkt_app_send_pong in tools/pkt_mgr.c which reads
   930→                        // getloadavg + gopt.ids_idle + gopt.n_users +
   931→                        // gopt.ids_active_user). Internal gsstats package
   932→                        // collects them with graceful fallback.
   933→                        pong := gsstats.PongPayload()
   934→                        if err := conn.WriteMsg(gspkt.MsgPONG, pong); err != nil {
   935→                                fmt.Fprintf(os.Stderr, "gs-netcat: PONG write failed: %v\n", err)
   936→                        }
   937→                        if verbose {
   938→                                fmt.Fprintf(os.Stderr, "[in-band] PING received, PONG sent\n")
   939→                        }
   940→                } else {
   941→                        // We are the connector: this is a PONG (reply to our PING).
   942→                        if verbose {
   943→                                if len(ev.Data) >= 5 {
   944→                                        load := uint16(ev.Data[0])<<8 | uint16(ev.Data[1])
   945→                                        idle := uint16(ev.Data[2])<<8 | uint16(ev.Data[3])
   946→                                        n := ev.Data[4]
   947→                                        user := ""
   948→                                        if len(ev.Data) >= 16 {
   949→                                                end := 5 + 11
   950→                                                if end > len(ev.Data) {
   951→                                                        end = len(ev.Data)
   952→                                                }
   953→                                                user = string(trimNUL(ev.Data[5:end]))
   954→                                        }
   955→                                        fmt.Fprintf(os.Stderr, "[in-band] PONG received (load=%d idle=%d users=%d user=%q)\n",
   956→                                                load, idle, n, user)
   957→                                } else {
   958→                                        fmt.Fprintf(os.Stderr, "[in-band] PONG received (short)\n")
   959→                                }
   960→                        }
   961→                }
   962→        case gspkt.MsgLOG:
   963→                // struct _pkt_app_log: type(1) + msg[63]
   964→                if len(ev.Data) >= 1 {
   965→                        lt := ev.Data[0]
   966→                        msg := ""
   967→                        if len(ev.Data) > 1 {
   968→                                msg = string(trimNUL(ev.Data[1:]))
   969→                        }
   970→                        fmt.Fprintf(os.Stderr, "[in-band] LOG(type=%d): %s\n", lt, msg)
   971→                }
   972→        case gspkt.MsgSTATUS:
   973→                if len(ev.Data) >= 1 {
   974→                        st := ev.Data[0]
   975→                        msg := ""
   976→                        if len(ev.Data) > 1 {
   977→                                msg = string(trimNUL(ev.Data[1:]))
   978→                        }
   979→                        fmt.Fprintf(os.Stderr, "[in-band] STATUS(type=%d): %s\n", st, msg)
   980→                        // GS_PKT_APP_STATUS_TYPE_NOPTY = 0x01: server couldn't
   981→                        // allocate a PTY. The C original calls stty_switch_nopty()
   982→                        // to relax raw mode (re-enable OPOST so newlines work).
   983→                        // We just log it — the user can pipe commands manually.
   984→                        // Since v0.1.4 the server includes the Spawn error
   985→                        // message in the STATUS payload so we can print WHY
   986→                        // the PTY failed (helps diagnose ConPTY/forkpty issues).
   987→                        if st == 0x01 {
   988→                                if msg != "" {
   989→                                        fmt.Fprintf(os.Stderr, "[gs-netcat] remote has no PTY — interactive shell unavailable (%s)\n", msg)
   990→                                } else {
   991→                                        fmt.Fprintf(os.Stderr, "[gs-netcat] remote has no PTY — interactive shell unavailable\n")
   992→                                }
   993→                        }
   994→                }
   995→        case gspkt.MsgWSIZE:
   996→                if len(ev.Data) == 4 {
   997→                        col := uint16(ev.Data[0])<<8 | uint16(ev.Data[1])
   998→                        row := uint16(ev.Data[2])<<8 | uint16(ev.Data[3])
   999→                        if conn.Role() == gsrn.RoleServer && ctx != nil && ctx.pty != nil {
  1000→                                // Apply the new size to the PTY. The shell sees
  1001→                                // a SIGWINCH and informs its child processes.
  1002→                                if err := ctx.pty.SetWindowSize(col, row); err != nil {
  1003→                                        if verbose {
  1004→                                                fmt.Fprintf(os.Stderr, "[in-band] WSIZE cols=%d rows=%d apply failed: %v\n", col, row, err)
  1005→                                        }
  1006→                                } else if verbose {
  1007→                                        fmt.Fprintf(os.Stderr, "[in-band] WSIZE applied: cols=%d rows=%d\n", col, row)
  1008→                                }
  1009→                        } else {
  1010→                                if verbose {
  1011→                                        fmt.Fprintf(os.Stderr, "[in-band] WSIZE: cols=%d rows=%d (no PTY to apply)\n", col, row)
  1012→                                }
  1013→                        }
  1014→                }
  1015→        case gspkt.MsgIDS:
  1016→                // Client is interested in IDS log messages. Subscribe
  1017→                // this conn to the process-wide gsids.Manager, which
  1018→                // polls utmp every 5s and sends MsgLOG (type=ALERT)
  1019→                // when new logins or idle→active transitions are
  1020→                // detected. Mirrors pkt_app_cb_ids in tools/pkt_mgr.c
  1021→                // + cbe_ids in tools/event_mgr.c + GS_IDS_utmp in
  1022→                // tools/ids.c.
  1023→                //
  1024→                // Subscribe returns an unsubscribe func; we register
  1025→                // it via conn.OnClose so it's called when the conn
  1026→                // terminates (the event-pump goroutine exits shortly
  1027→                // after, so there's no race).
  1028→                if conn.Role() == gsrn.RoleServer {
  1029→                        unsub := gsids.Subscribe(conn)
  1030→                        conn.OnClose(func() {
  1031→                                unsub()
  1032→                                if verbose {
  1033→                                        fmt.Fprintf(os.Stderr, "[in-band] IDS subscription removed\n")
  1034→                                }
  1035→                        })
  1036→                        if verbose {
  1037→                                fmt.Fprintf(os.Stderr, "[in-band] IDS request received — subscribed to utmp events\n")
  1038→                        }
  1039→                } else if verbose {
  1040→                        fmt.Fprintf(os.Stderr, "[in-band] IDS request received (ignored — not server role)\n")
  1041→                }
  1042→        case gspkt.MsgPWD:
  1043→                // Server: reply with ChnPWD carrying the shell's cwd.
  1044→                if conn.Role() == gsrn.RoleServer && ctx != nil && ctx.pid > 0 {
  1045→                        wd := readShellCwd(ctx.pid)
  1046→                        if wd == "" {
  1047→                                wd, _ = os.Getwd()
  1048→                        }
  1049→                        if err := conn.WritePWDReply(wd); err != nil {
  1050→                                fmt.Fprintf(os.Stderr, "gs-netcat: PWD reply write failed: %v\n", err)
  1051→                        } else if verbose {
  1052→                                fmt.Fprintf(os.Stderr, "[in-band] PWD request, replied with %q\n", wd)
  1053→                        }
  1054→                } else if verbose {
  1055→                        fmt.Fprintf(os.Stderr, "[in-band] PWD request received (no PTY/shell to query)\n")
  1056→                }
  1057→        default:
  1058→                if verbose {
  1059→                        fmt.Fprintf(os.Stderr, "[in-band] unhandled msg type %d (%d bytes)\n", ev.Type, len(ev.Data))
  1060→                }
  1061→        }
  1062→}

  1064→func handleChannel(conn *gsconn.Conn, ctx *eventCtx, ev gspkt.Event, verbose bool) {
  1065→        switch ev.Type {
  1066→        case gspkt.ChnPWD:
  1067→                // Server's working-directory reply (NUL-terminated string).
  1068→                msg := string(trimNUL(ev.Data))
  1069→                fmt.Fprintf(os.Stderr, "[in-band] remote wd: %s\n", msg)
  1070→        default:
  1071→                if verbose {
  1072→                        fmt.Fprintf(os.Stderr, "[in-band] unhandled chn %d (%d bytes)\n", ev.Type, len(ev.Data))
  1073→                }
  1074→        }
  1075→}

  1077→// readProcCwd / readShellCwd lives in cwd_linux.go / cwd_other.go.
  1078→// Kept here only as a reference; trimNUL is the only generic helper.
  1079→
  1080→// trimNUL returns b truncated at the first NUL byte (if any), with
  1081→// trailing whitespace trimmed.
  1082→func trimNUL(b []byte) []byte {
  1083→        for i, c := range b {
  1084→                if c == 0 {
  1085→                        b = b[:i]
  1086→                        break
  1087→                }
  1088→        }
  1089→        // trim trailing whitespace
  1090→        for len(b) > 0 && (b[len(b)-1] == ' ' || b[len(b)-1] == '\n' || b[len(b)-1] == '\r' || b[len(b)-1] == '\t') {
  1091→                b = b[:len(b)-1]
  1092→        }
  1093→        return b
  1094→}

  1096→// manPageText returns the text printed by `-m`. Mirrors the layout of the
  1097→// upstream man page while documenting the Go-port-specific behaviour
  1098→// and the subset of flags currently implemented.
  1099→//
  1100→// The text is stored XOR-encrypted in manpage.enc (embedded via go:embed)
  1101→// to avoid leaving plaintext help strings in the binary's .rodata. Plain
  1102→// base64 encoding is NOT sufficient: modern AV scanners (including MS
  1103→// Defender) automatically decode base64 blobs during static analysis and
  1104→// would rediscover the original "TLS-SRP", "cmd.exe", "/etc/gs_secret"
  1105→// strings. go:embed assets are stored as-is by the Go toolchain and are
  1106→// NOT obfuscated by garble -literals, so a multi-byte XOR layer is
  1107→// mandatory. garble -literals DOES obfuscate the manPageKey byte slice,
  1108→// raising the bar beyond simple single-byte XOR brute-force.
  1109→//
  1110→//go:embed manpage.enc
  1111→var manPageEnc []byte
  1112→
  1113→var manPageKey = []byte{0x34, 0x62, 0x75, 0x33, 0xb9, 0xa7, 0x83, 0x7f, 0xd7, 0x45, 0x90, 0x8f, 0x97, 0xa7, 0x7f, 0x73}
  1114→
  1115→func manPageText() string {
  1116→        out := make([]byte, len(manPageEnc))
  1117→        for i, b := range manPageEnc {
  1118→                out[i] = b ^ manPageKey[i%len(manPageKey)]
  1119→        }
  1120→        return string(out)
  1121→}
